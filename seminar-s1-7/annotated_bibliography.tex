\section{Core Reading (Tài Liệu Cốt Lõi)}
\begin{enumerate}[leftmargin=1.4em, itemsep=6pt, topsep=3pt]
  \item \textbf{Anthropic. (Dec 2024).} \textit{Building Effective Agents.} Anthropic Research Whitepaper.\\
  \textbf{Phạm vi trọng tâm:} Phân biệt Workflows vs. Autonomous Agents; các khối nguyên tử (prompt chaining, routing, parallelization, orchestrator-workers, evaluator-optimizer); triết lý ưu tiên sự đơn giản (simplicity-first).\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Dù lập luận thuyết phục rằng các workflow tiền định vượt trội hơn hẳn agent tự do trong môi trường doanh nghiệp có cấu trúc, tài liệu lại đánh giá quá nhẹ tính dễ gãy vỡ (fragility) của các ngưỡng phân nhánh cố định và hoàn toàn bỏ ngỏ cơ chế phục hồi trạng thái khi một mắt xích trong chuỗi gặp sự cố âm thầm.
\end{enumerate}

\section{Extension Readings (Các Tài Liệu Mở Rộng Cho Nhóm Thuyết Trình)}
\begin{enumerate}[leftmargin=1.4em, itemsep=6pt, topsep=3pt, start=2]
  \item \textbf{Anthropic. (Feb 2025).} \textit{How We Built Our Multi-Agent Research System.} Anthropic Engineering Blog.\\
  \textbf{Phạm vi trọng tâm:} Mô hình phân cấp Lead/Subagents; cô lập ngữ cảnh (context isolation); đánh đổi thực tế về chi phí token gấp $3\times$--$8\times$ và độ trễ.\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Việc chia nhỏ tác vụ cho các subagent giải quyết triệt để vấn đề thoái hóa ngữ cảnh nhưng phải trả giá bằng sự bùng nổ token gấp $3\times$--$8\times$ và nút thắt cổ chai độ trễ tại Lead Agent khi phải tổng hợp kết quả mà thường xuyên bỏ sót các mâu thuẫn tinh vi giữa các tác tử phụ.

  \item \textbf{OpenAI. (Feb 2025).} \textit{A Practical Guide to Building Agents.} OpenAI Developer Documentation.\\
  \textbf{Phạm vi trọng tâm:} Vòng lặp gọi công cụ do mô hình điều hướng (model-directed tool calling); cơ chế rào chắn (guardrails); khung đánh giá (evals); chuyển giao tác vụ (handoffs).\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Tài liệu cổ xúy việc trao quyền tự trị cho vòng lặp gọi công cụ của mô hình nhưng né tránh thực tế rằng xác suất tích lũy lỗi ($P = p^n$) sẽ nhanh chóng làm suy giảm nghiêm trọng độ tin cậy trong các bài toán pháp lý/tài chính nếu thiếu validator xác thực ground-truth độc lập.

  \item \textbf{LangChain Team. (2024--2025).} \textit{LangGraph: Multi-Agent Workflows as Cyclic State Graphs.} Technical Documentation.\\
  \textbf{Phạm vi trọng tâm:} Đồ thị trạng thái có chu trình (cyclic state graphs); thực thi bền bỉ (durable execution); lưu trữ checkpoint; phục hồi lỗi; can thiệp của con người (human-in-the-loop).\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Cơ chế Checkpointing giải quyết hiệu quả bài toán thực thi bền bỉ và tua lại trạng thái khi gặp lỗi, nhưng việc cưỡng bức mọi hành vi suy luận thích ứng vào một máy trạng thái hữu hạn cứng nhắc làm tăng vọt chi phí bảo trì schemas và khiến hệ thống quay về bản chất của một luồng nghiệp vụ truyền thống.
\end{enumerate}
