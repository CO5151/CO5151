\section{Core Papers (Các Công Trình Nền Tảng)}
\begin{enumerate}[leftmargin=1.4em, itemsep=2.5pt, topsep=1pt]
  \item \textbf{Wu, Q., et al. (Microsoft Research, 2023).} \textit{AutoGen: Enabling Next-Gen LLM Applications via Multi-Agent Conversation.} arXiv:2308.08155.\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Cơ chế hội thoại tự do giữa các tác tử dễ dẫn đến bùng nổ token phi kiểm soát và hiện tượng nịnh bợ (sycophancy) tập thể, khiến chi phí điều phối vượt xa giá trị giải thuật mang lại nếu thiếu một máy trạng thái hữu hạn (FSM) cưỡng chế điều kiện dừng.

  \item \textbf{Anthropic. (Dec 2024).} \textit{Building Effective Agents.} Anthropic Research Whitepaper.\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Dù phân lập chuẩn xác các khối nguyên tử (building blocks) thực dụng, tài liệu đã đánh giá thấp sự phức tạp của việc phục hồi trạng thái khi một mắt xích trong chuỗi công cụ thất bại âm thầm (silent truncation) giữa môi trường sản xuất thực tế.

  \item \textbf{Shinn, N., Cassano, F., Labash, B., et al. (NeurIPS 2023).} \textit{Reflexion: Language Agents with Verbal Reinforcement Learning.}\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Khả năng tự phản biện qua bộ nhớ ngôn ngữ bị phụ thuộc nặng nề vào năng lực suy luận của chính mô hình đó, dẫn đến bẫy tự xác nhận (self-confirmation bias) khi tác tử kiên quyết dùng các chu kỳ suy luận sau để hợp thức hóa một tiên đề sai ban đầu.

  \item \textbf{Madaan, A., Tandon, N., Gupta, P., et al. (NeurIPS 2023).} \textit{Self-Refine: Iterative Refinement with Self-Feedback.}\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Quá trình tinh chỉnh lặp lại không có tín hiệu phản hồi khách quan từ môi trường ngoài (ground-truth execution) chỉ làm thay đổi phong cách biểu đạt văn phong chứ không cải thiện độ chính xác bản chất của nghiệm logic.
\end{enumerate}

\section{Extension \& Methodological Papers (Công Trình Mở Rộng \& Phương Pháp Luận)}
\begin{enumerate}[leftmargin=1.4em, itemsep=2.5pt, topsep=1pt, start=5]
  \item \textbf{Hong, S., Zheng, X., Chen, J., et al. (ICLR 2024).} \textit{MetaGPT: Meta Programming for Multi-Agent Collaborative Framework.}\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Việc áp đặt quy trình thao tác chuẩn (SOP) tĩnh kiểu kỹ nghệ phần mềm công nghiệp làm tê liệt khả năng thích ứng linh hoạt của LLM trước các sự cố ngoại lệ phát sinh ngoài kịch bản dự kiến.

  \item \textbf{Gou, Z., Shao, Z., Gong, Y., et al. (ICLR 2024).} \textit{CRITIC: Large Language Models Can Self-Correct with Tool-Interactive Critiquing.}\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Dù chứng minh được vai trò sống còn của công cụ kiểm chứng ngoài, nghiên cứu né tránh bài toán bùng nổ độ trễ tương tác (latency overhead) khi mỗi bước phản biện đều phải đợi lời gọi API từ bên thứ ba.

  \item \textbf{Chen, W., et al. (2024).} \textit{Are More LLM Calls All You Need? Towards a Principled Understanding of Scaling Agentic Workflows.} arXiv:2403.02419.\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Công trình bóc trần luận điểm rằng phần lớn sự vượt trội của kiến trúc đa tác tử thực chất chỉ là hệ quả của việc gia tăng ngân sách tính toán (sampling compute), và hiệu năng này suy giảm rất nhanh theo quy luật lợi suất giảm dần (diminishing returns).

  \item \textbf{LangChain Team. (2024).} \textit{LangGraph: Multi-Agent Workflows as Cyclic Graphs.} Technical Documentation.\\
  \textcolor{critiquecolor}{\textbf{Nhận xét phản biện:}} Việc mô hình hóa agent thành đồ thị trạng thái xác định giải quyết triệt để tính phi tất định nhưng thực chất đã biến hệ thống thành một luồng công việc kỹ thuật truyền thống hơn là một thực thể trí tuệ nhân tạo tự trị.
\end{enumerate}
