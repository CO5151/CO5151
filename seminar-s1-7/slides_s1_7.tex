% SLIDE 5: WORKFLOWS VS AUTONOMOUS AGENTS
\begin{frame}{Workflows vs. Autonomous Agents: Kiểm Soát Luồng Thực Thi}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Luồng cố định bằng code vs. Luồng do mô hình tự quyết định}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Workflows: Luồng Code Cố Định}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Control Flow:} Hard-coded bằng code truyền thống (Python, Go).
          \item \textbf{Vai trò của LLM:} Bước suy luận nguyên tử (atomic step) bên trong pipeline định sẵn.
          \item \textbf{Routing:} Logic if-else, kiểm tra schema xác định, chuyển đổi trạng thái tiền định.
          \item \textbf{Ưu điểm cốt lõi:} Độ dự đoán cao, tái lập 100\%, latency thấp, token overhead tối thiểu ($1.0\times$).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Autonomous Agents: Luồng Do Mô Hình Tự Quyết Định}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Control Flow:} Do vòng lặp suy luận của LLM quyết định (ReAct loop, Tool Calling).
          \item \textbf{Vai trò của LLM:} Tự chọn gọi tool nào, khi nào dừng lại và rẽ nhánh về đâu.
          \item \textbf{Routing:} Hoàn toàn động dựa trên quan sát (observations) ở runtime.
          \item \textbf{Đánh đổi lớn:} Linh hoạt với bài toán mở, nhưng dễ tích lũy sai số và bùng nổ chi phí.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Định Nghĩa Chuẩn Xác Của Anthropic (2024)}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textit{"Workflows are systems where LLMs and tools are orchestrated through predefined code paths. Agents, on the other hand, are systems where LLMs dynamically direct their own processes and tool usage."}
  \end{tcolorbox}
\end{frame}

% SLIDE 6: VÌ SAO ANTHROPIC KHUYÊN BỎ AGENT
\begin{frame}{Vì Sao Anthropic Khuyến Cáo Ưu Tiên Workflow Thay Vì Agent?}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Hiện tượng tích lũy sai số ($P = p^n$), bùng nổ token và rủi ro vận hành}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{Hiện Tượng Tích Lũy Sai Số ($P = p^n$)}}
        \footnotesize
        Giả sử mỗi bước gọi tool mô hình đạt độ chính xác $p \in [0.90, 0.95]$:
        \begin{itemize}\scriptsize
          \item \textbf{Workflow 3 bước:} $P_{\text{success}} = 0.95^3 \approx \mathbf{85.7\%}$
          \item \textbf{Vòng lặp agent 8 bước:} $P_{\text{success}} = 0.95^8 \approx \mathbf{66.3\%}$
          \item \textbf{Vòng lặp agent 15 bước:} $P_{\text{success}} = 0.95^{15} \approx \mathbf{46.3\%}$
        \end{itemize}
        \vspace{0.1cm}
        \alert{\textbf{Phát hiện cốt lõi:}} Trong unconstrained agent loop, độ tin cậy của toàn hệ thống giảm theo exponential ($P = p^n$) khi số bước thực thi (trajectory steps) tăng.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Bốn Rủi Ro Khi Lạm Dụng Quyền Tự Trị Quá Sớm}}
        \footnotesize
        \begin{enumerate}
          \item \textbf{Độ trễ không giới hạn (Unbounded Latency):} Số turns do LLM quyết định; thời gian phản hồi biến thiên khôn lường.
          \item \textbf{Chi phí Token bùng nổ:} Lịch sử hội thoại tích lũy liên tục gây quadratic prompt bloat.
          \item \textbf{Bế tắc khi Debug:} Hành vi non-deterministic khiến việc tái hiện edge-case bugs cực kỳ khó khăn.
          \item \textbf{Trôi lệch ngữ cảnh (Context Drift):} Phản hồi tool rác làm bẩn bộ nhớ, gây ra ảo giác dây chuyền.
        \end{enumerate}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Luận Điểm Trọng Tâm Trả Lời Đề Cương}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textbf{Vì sao Anthropic khuyến cáo không dùng agent cho phần lớn bài toán?} Vì đối với 90\% ứng dụng thực tế, workflows đem lại độ tin cậy cao hơn, độ trễ thấp hơn, chi phí rẻ hơn 3--8 lần và dễ debug hơn nhiều so với autonomous agents.
  \end{tcolorbox}
\end{frame}

% SLIDE 7: PROMPT CHAINING & ROUTING
\begin{frame}{Mẫu Thiết Kế Cơ Bản: Prompt Chaining và Routing}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Nguyên lý hoạt động, ưu điểm kỹ thuật và chi phí đánh đổi}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Mẫu 1: Prompt Chaining}}
        \footnotesize
        \textbf{Cơ chế:} Chia nhỏ bài toán phức tạp thành chuỗi các lệnh gọi LLM tuần tự; bước sau tiêu thụ đầu ra của bước trước.\\
        \vspace{0.1cm}
        \textbf{Lợi ích mang lại (What It Buys):}
        \begin{itemize}\scriptsize
          \item Giảm tải áp lực nhận thức (cognitive load) trên mỗi prompt.
          \item Cho phép chèn programmatic validation giữa các bước.
          \item Dùng mô hình nhỏ, rẻ hơn cho các bước trung gian.
        \end{itemize}
        \textbf{Chi phí đánh đổi (What It Costs):}
        \begin{itemize}\scriptsize
          \item Độ trễ tăng tuyến tính theo số bước ($T = \sum t_i$).
          \item Kém linh hoạt: Không tự phục hồi nếu bước trước sinh lỗi mà không có explicit branching code.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Mẫu 2: Routing}}
        \footnotesize
        \textbf{Cơ chế:} Bộ phân loại (LLM hoặc regex/embedding) nhận diện input intent để dẫn hướng tới nhánh xử lý chuyên biệt.\\
        \vspace{0.1cm}
        \textbf{Lợi ích mang lại (What It Buys):}
        \begin{itemize}\scriptsize
          \item Chuyên môn hóa: Mỗi nhánh có system prompt và tools hẹp, tối ưu riêng.
          \item Ngăn ngừa prompt bloat (tránh nhồi 50 tools vào 1 prompt).
          \item Tối ưu chi phí: Route các truy vấn đơn giản sang small models.
        \end{itemize}
        \textbf{Chi phí đánh đổi (What It Costs):}
        \begin{itemize}\scriptsize
          \item Bộ phân loại router đoán sai sẽ làm hỏng toàn bộ pipeline con.
          \item Chi phí bảo trì: Thêm nhánh mới đòi hỏi cập nhật cả router và downstream handlers.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 8: EVALUATOR-OPTIMIZER PATTERN
\begin{frame}{Mẫu Thiết Kế Phản Hồi: Evaluator-Optimizer}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Tinh chỉnh lặp, chi phí đánh đổi và bẫy nịnh bợ (Sycophancy)}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{Cơ Chế Evaluator-Optimizer}}
        \footnotesize
        \textbf{Hai vai trò tách biệt rõ rệt:}
        \begin{enumerate}\scriptsize
          \item \textbf{Generator LLM:} Tạo ra candidate output dựa trên yêu cầu ban đầu.
          \item \textbf{Evaluator (LLM hoặc Tool):} Đánh giá output dựa trên tiêu chí rõ ràng và trả về structured feedback.
          \item \textbf{Refinement Loop:} Generator điều chỉnh lại output dựa trên feedback cho đến khi đạt chuẩn.
        \end{enumerate}
        \vspace{0.1cm}
        \textbf{Lợi ích mang lại (What It Buys):}
        \begin{itemize}\scriptsize
          \item Cải thiện output quality rõ rệt trong code generation, translation, và formal drafting.
          \item Mô phỏng quy trình drafting và peer-review của con người.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Chi Phí Đánh Đổi \& Các Dạng Lỗi Thường Gặp}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Bùng nổ Token:} $K$ vòng lặp làm tăng token tiêu thụ lên $K \times$, độ trễ kéo dài tương ứng.

          \item \textbf{Giải pháp:} Bắt buộc tích hợp \textbf{external deterministic tools} (linters, unit tests, API checkers) thay vì pure LLM self-reflection.
        \end{itemize}
      \end{alertblock}

\begin{frame}{Mẫu Orchestrator-Workers vs. Cửa Sổ Ngữ Cảnh Lớn (Long Context)}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Hạn Chế Của Long Context Window (2M Tokens)}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Thoái hóa chú ý ("Lost in the Middle"):} Khi context vượt quá 30k tokens, attention score trên các đoạn ở giữa suy giảm rõ rệt.
          \item \textbf{Ô nhiễm ngữ cảnh (Context Pollution):} Intermediate scratchpads và stale tool outputs làm nhiễu prompt, kích hoạt hallucinations.
          \item \textbf{Độ trễ và chi phí bậc hai (Quadratic Cost):} Xử lý 100k+ tokens ở mọi turn khiến execution chậm và cực kỳ đắt đỏ.
          \item \textbf{Sụp đổ toàn bộ (All-or-Nothing Failure):} Nếu step 7 crash, toàn bộ context phải re-process từ đầu.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Giải Pháp Orchestrator-Workers}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Lập kế hoạch động (Dynamic Planning):} Orchestrator phân rã bài toán và khởi tạo specialized workers theo nhu cầu.
          \item \textbf{Cô lập ngữ cảnh (Context Isolation):} Mỗi worker \textbf{chỉ nhận đúng thông tin phục vụ subtask} của nó.
          \item \textbf{Thực thi song song (Parallel Execution):} Independent workers chạy song song (tra cứu vốn điều lệ song song với trần sở hữu ngoại).
          \item \textbf{Bàn giao tổng hợp (Synthesized Handoff):} Workers trả về concise structured summaries cho Orchestrator, giữ top-level context luôn sạch.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{LangGraph: Điều Phối Đa Tác Tử Bằng Cyclic State Graphs}

      \begin{block}{\textbf{Các Khái Niệm Cốt Lõi Của LangGraph}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Centralized State Schema:} Shared, typed state dictionary được truyền xuyên suốt qua tất cả các nodes.
          \item \textbf{State Reducers:} Functions quy định cách node updates merge vào global state (e.g., append messages vs. overwrite).
          \item \textbf{Nodes as Execution Units:} Mỗi node là một LLM call, một isolated subagent, hoặc một deterministic tool.
          \item \textbf{Conditional Edges:} Python functions kiểm tra state để route execution dynamically (nếu audit fail $\to$ route to recovery).
        \end{itemize}

        \footnotesize

          \item \textbf{DAG truyền thống chỉ đi một chiều:} Nếu retrieval lấy phải revoked statute, DAG không thể quay ngược lại (backtrack).
          \item \textbf{Đồ thị chu trình cho phép vòng lặp:} Graph có thể route back về planning node để tìm văn bản thay thế kèm retry counter.
          \item \textbf{Vòng lặp có chặn trên (Bounded Loops):} Graph edges thực thi max retry limits (FSM circuit breaker) để triệt tiêu hoàn toàn infinite loops.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

    \begin{column}{0.46\textwidth}
      \begin{alertblock}{\textbf{Cơ Sở Kỹ Thuật Của Cô Lập Ngữ Cảnh (Context Isolation)}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Triệt tiêu bùng nổ token bậc hai:} Thay vì nhồi monolithic context khổng lồ ($>30\text{k}$ tokens), mỗi worker chạy trong context sạch $<1,500$ tokens $\to$ Giảm $70-80\%$ attention noise.
          \item \textbf{Chống thoái hóa chú ý (Lost in the Middle):} Tập trung attention weights tối đa vào đúng nhiệm vụ chuyên môn của từng subagent.
          \item \textbf{Khoanh vùng lỗi (Fault Containment):} Kết quả rác hoặc văn bản hết hiệu lực ở retrieval node bị cô lập hoàn toàn, không lây nhiễm vào drafter.
        \end{itemize}
      \end{alertblock}
      \vspace{0.08cm}
      \begin{block}{\scriptsize \textbf{Minh Họa Bằng Kiến Trúc Hệ Thống G4}}
        \tiny Triển khai thực tế trong nhóm: \texttt{LegalOrchestrator} điều phối song song \texttt{LawGraphAgent} (Retrieval) và \texttt{ClaimAuditorAgent} (Audit Oracle) trước khi chuyển giao cho \texttt{DrafterAgent}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 17: AGENT DESIGN PATTERNS: ARCHITECTURAL SELECTION & PRUNING
\begin{frame}{Lựa Chọn \& Cắt Tỉa (Pruning) Kiến Trúc Agent}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Vì sao vòng lặp ReAct tự do thất bại trong các bài toán yêu cầu độ tin cậy cao}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Mẫu Kiến Trúc Bị Loại Bỏ: Vòng Lặp ReAct Tự Do}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Bản chất:} Một LLM duy nhất tự do gọi tools theo chu trình lặp Thought $\to$ Action $\to$ Observation.
          \item \textbf{3 Điểm nghẽn lỗi hệ thống trong Production:}
                \begin{enumerate}\scriptsize
                  \item \textbf{Sai số tích lũy cấp số nhân ($P = \prod p_i$):} Với $p = 0.92$, sau 6 bước $P \approx 60\%$, sau 12 bước $P \approx 36\%$ $\to$ Quá mỏng manh cho các bài toán quan trọng.
                  \item \textbf{Bùng nổ chi phí và độ trễ bậc hai:} Lịch sử hội thoại dài thêm sau mỗi turn khiến tổng token tiêu thụ tăng $O(N^2)$, gây lãng phí lớn.
                  \item \textbf{Zero Fault Containment:} Tiến trình nằm trên RAM volatile; lỗi mạng hoặc timeout ở turn cuối làm sụp đổ toàn bộ pipeline từ đầu.
                \end{enumerate}
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Mẫu Kiến Trúc Được Chọn: Structured Orchestrator-Workers}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Bản chất:} Phân rã bài toán thành đồ thị trạng thái có hướng với quyền tự trị bị chặn biên (Bounded Autonomy).
          \item \textbf{3 Lợi thế kiến trúc vượt trội:}
                \begin{enumerate}\scriptsize
                  \item \textbf{Khoanh vùng sai số:} Worker chỉ có thẩm quyền trong phạm vi hẹp; lỗi của một tác tử bị chặn đứng tại boundary, không lan truyền.
                  \item \textbf{Thực thi bền bỉ (Durable Execution):} Tích hợp checkpointing lưu trữ trạng thái sau từng bước $\to$ Cho phép phục hồi hoặc retry cục bộ.
                  \item \textbf{Kiểm soát ngân sách token \& thời gian:} Áp đặt ngân sách token và số lượt suy luận (max turns) cứng trên từng node, loại bỏ vòng lặp vô tận.
                \end{enumerate}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Bài Học Quyết Định Kiến Trúc Trong Thực Tế}},fonttitle=\footnotesize\bfseries]
    \footnotesize Quyền tự trị không ràng buộc (unconstrained autonomy) là nguồn gốc của các lỗi bất định trong doanh nghiệp. Thiết kế chuẩn mực là kết hợp \textbf{khả năng phân rã linh hoạt của LLM với luồng kiểm soát trạng thái tiền định bằng mã nguồn}.
  \end{tcolorbox}
\end{frame}

% SLIDE 18: FAULT-TOLERANT ARCHITECTURE: DETERMINISTIC GROUND-TRUTH ORACLES
\begin{frame}{Thiết Kế Hệ Thống Chịu Lỗi: Deterministic Oracles vs. LLM Self-Critique}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Tách biệt tầng suy luận khỏi tầng kiểm định bằng các công cụ tiền định độc lập}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{\textbf{Hạn Chế Khi Để LLM Tự Đánh Giá (LLM-as-a-Judge)}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Vòng lặp phân phối tham số (Parametric Circularity):} Yêu cầu LLM "tự rà soát" (\textit{"Are you sure?"}) chỉ tái lấy mẫu từ cùng một phân phối $P_\theta(y \vert x)$ đã sinh lỗi ban đầu $\to$ Không thể phát hiện ảo giác bằng chính mô hình tạo ra ảo giác.
          \item \textbf{Bẫy nịnh bợ (Sycophancy Trap):} Evaluator LLM có xu hướng thiên vị và đồng thuận với candidate output của Generator, đặc biệt khi chia sẻ cùng base model.
          \item \textbf{Điểm mù tri thức thời gian thực:} LLM không thể tự biết một văn bản quy phạm pháp luật đã bị bãi bỏ vào ngày hôm qua nếu không truy vấn live external registry.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Cơ Chế Kiểm Định Bằng Deterministic Oracles}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Tách rời hai mặt phẳng (Decoupled Planes):} Tách biệt tầng suy luận xác suất (probabilistic reasoning) khỏi tầng kiểm định tiền định (deterministic validation).
          \item \textbf{Cổng kiểm tra nhị phân (Binary Assertion Gates):} Sử dụng các công cụ xác định (Linters, Type Checkers, Schema Validators, Database Registry APIs) trả về kết quả nhị phân: \texttt{VALID} hoặc \texttt{INVALID}.
          \item \textbf{Ngắt mạch cứng (Hard Circuit Breakers):} Khi Oracle phát hiện vi phạm, đồ thị lập tức ngắt luồng và kích hoạt nhánh phục hồi có điều kiện.
          \item \textbf{Nhật ký kiểm toán bất biến (Audit Trail):} Lưu trữ bằng chứng nhị phân vào durable state phục vụ giải trình kỹ thuật và pháp lý.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 19: STATE PERSISTENCE & RESILIENT GRAPH ROLLBACKS
\begin{frame}{Xử Lý Lỗi \& Cơ Chế Rollback Trạng Thái Trong Đồ Thị Thực Thi}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Kiểm soát lỗi từng phần và quay lui trạng thái dựa trên Checkpointing}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{Trình Tự Thực Thi Đồ Thị \& Cơ Chế Rollback Vi Sai}}
        \footnotesize
        \begin{enumerate}\scriptsize
          \item \textbf{Checkpoint $C_1$ (Lập kế hoạch):} Orchestrator phân rã truy vấn thành Subgoal 1 (Điều kiện cấp phép) và Subgoal 2 (Tỷ lệ sở hữu nước ngoài) $\to$ Lưu snapshot $C_1$ (SQLite/Postgres).
          \item \textbf{Thực thi song song:} Worker 1 tra cứu Subgoal 1; Worker 2 tra cứu Subgoal 2 $\to$ Lưu checkpoint $C_2$.
          \item \textbf{Phát hiện lỗi bằng Oracle:} Auditor Oracle phát hiện văn bản của Subgoal 1 đã hết hiệu lực $\to$ Kích hoạt Conditional Failure Edge.
          \item \textbf{Differential Rollback (Quay lui vi sai):}
                $\bullet$ Khôi phục trạng thái đồ thị về $C_1$.\\
                $\bullet$ \textbf{Bảo toàn dữ liệu hợp lệ:} Giữ nguyên kết quả Subgoal 2 trong State Cache (không tính toán lại!).\\
                $\bullet$ Chỉ tái điều phối Subgoal 1 sang văn bản kế thừa hợp lệ kèm bộ đếm retry ($K \le 2$).
          \item \textbf{Hội tụ trạng thái:} Drafter tổng hợp hồ sơ hoàn chỉnh từ dữ liệu đã qua xác thực $100\%$ $\to$ Lưu checkpoint $C_3$.
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Đặc Tính Đảm Bảo Của Hệ Thống}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Không bao giờ Cold Restart:} Tiết kiệm tối đa chi phí token và thời gian bằng cách không chạy lại các subtasks hợp lệ đã hoàn thành.
          \item \textbf{Tiếp tục thực thi xác định (Resilient Resumption):} Khi gặp sự cố mạng hoặc sập container, runtime tiếp tục mượt mà từ checkpoint an toàn gần nhất.
          \item \textbf{Gỡ lỗi Time-Travel (Time-Travel Debugging):} Cho phép kỹ sư phân nhánh trạng thái từ bất kỳ checkpoint nào để tái lập và chẩn đoán lỗi.
        \end{itemize}
      \end{alertblock}
      \vspace{0.08cm}
      \begin{block}{\scriptsize \textbf{Minh Họa Bằng Trace Thực Nghiệm Trong Demo}}
        \tiny Trace thực nghiệm xử lý chuyển tiếp Nghị định 101/2012 (bãi bỏ) sang Nghị định 52/2024 trong mã nguồn \texttt{LegalOrchestrator} của nhóm.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 20: ORCHESTRATION TOPOLOGIES & COMMUNICATION ARCHITECTURES
\begin{frame}{So Sánh Các Mô Hình Điều Phối: Centralized, Swarms \& Cyclic Graphs}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Đánh giá ba kiến trúc điều phối: Hub-and-Spoke, Decentralized Swarms và State Graphs}

  \begin{columns}[t]
    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{1. Hub-and-Spoke (Anthropic)}}
        \footnotesize
        \textbf{Kiến trúc hình sao tập trung:}
        \begin{itemize}\scriptsize
          \item Lead điều phối trung tâm; subagents hoạt động độc lập và báo cáo về Lead.
          \item \textbf{Ưu điểm:} Toàn quyền kiểm soát trạng thái; cô lập ngữ cảnh tuyệt đối giữa các subagents.
          \item \textbf{Đánh đổi:} Nút thắt độ trễ tại Lead (\textbf{Straggler Effect}: $\max t_i$); chi phí token tổng hợp cao ($3-8\times$).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{alertblock}{\textbf{2. Decentralized Swarms}}
        \footnotesize
        \textbf{Mạng ngang hàng (Peer-to-Peer):}
        \begin{itemize}\scriptsize
          \item Tác tử tự do handoff và giao tiếp ngang hàng trực tiếp (e.g. OpenAI Swarm).
          \item \textbf{Ưu điểm:} Tính linh hoạt rất cao trong đàm phán đa bên không cấu trúc.
          \item \textbf{Rủi ro chí mạng:} Bùng nổ tin nhắn $O(N^2)$; dễ kẹt vòng lặp ping-pong vô tận; không thể rollback trạng thái bền bỉ.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{3. Cyclic State Graphs}}
        \footnotesize
        \textbf{Kiến trúc LangGraph:}
        \begin{itemize}\scriptsize
          \item Điều phối qua đồ thị trạng thái với strict typed schema và conditional edges.
          \item \textbf{Ưu điểm:} Kiểm soát chặt chẽ chu trình lặp; hỗ trợ durable checkpointing và deterministic time-travel.
          \item \textbf{Đánh đổi:} Đòi hỏi định nghĩa state schema chặt chẽ từ đầu; tính tự do thấp hơn swarm.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Đúc Kết Về Lựa Chọn Mô Hình Điều Phối Doanh Nghiệp}},fonttitle=\footnotesize\bfseries]
    \footnotesize Các hệ thống doanh nghiệp nghiêm túc tuyệt đối tránh dùng decentralized swarms tự do. Tiêu chuẩn kiến trúc chiếm ưu thế hiện nay kết hợp \textbf{phân quyền Hub-and-Spoke} với \textbf{Đồ thị trạng thái chu trình (Cyclic State Graphs) để xử lý phục hồi lỗi}.
  \end{tcolorbox}
\end{frame}

\begin{frame}{Nguyên Lý Thiết Kế Hệ Thống Agent Trong Môi Trường Production}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Tổng hợp bài học kỹ thuật từ Anthropic, OpenAI và LangGraph}

  \begin{columns}[t]
    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{1. Nguyên Tắc Simplicity-First}}
        \footnotesize
        \begin{itemize}
          \item Mặc định ưu tiên prompt chaining và deterministic routing.
          \item Dùng code cho control flow và chỉ dùng LLM cho atomic reasoning.
          \item Không bao giờ đưa dynamic orchestration vào nếu fixed workflow đã đáp ứng.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{2. Cô Lập Ngữ Cảnh (Context Isolation)}}
        \footnotesize
        \begin{itemize}
          \item Khi bắt buộc dùng multi-agent, áp dụng mô hình Lead/Subagent của Anthropic.
          \item Không bao giờ chia sẻ một scratchpad chung lộn xộn giữa các agents.
          \item Chấp nhận token multiplier $3-8\times$ như cái giá của context hygiene.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{3. Thiết Kế Bền Bỉ (Durable Execution)}}
        \footnotesize
        \begin{itemize}
          \item Lưu trữ mọi bước chuyển trạng thái bằng immutable checkpointing.
          \item Tách rời kiểm định bằng external deterministic oracles.
          \item Đặt giới hạn số lần retry có chặn trên (circuit-breakers) cho các chu trình lặp.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.25cm}
  \begin{alertblock}{\textbf{Đúc Kết Nguyên Lý Kỹ Thuật Hệ Thống}}
