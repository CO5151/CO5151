% SLIDE 5: WORKFLOWS VS AUTONOMOUS AGENTS
\begin{frame}{Workflows vs. Autonomous Agents: Kiểm Soát Luồng Thực Thi}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Luồng cố định bằng code vs. Luồng do mô hình tự quyết định}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Workflows: Luồng Code Cố Định}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Control Flow:} Hard-coded bằng code truyền thống (Python, Go).
          \item \textbf{Vai trò của LLM:} Bước suy luận nguyên tử (atomic step) bên trong pipeline định sẵn.
          \item \textbf{Routing:} Logic if-else, kiểm tra schema xác định, chuyển đổi trạng thái tiền định.
          \item \textbf{Ưu điểm cốt lõi:} Độ dự đoán cao, tái lập 100\%, latency thấp, token overhead tối thiểu ($1.0\times$).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Autonomous Agents: Luồng Do Mô Hình Tự Quyết Định}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Control Flow:} Do vòng lặp suy luận của LLM quyết định (ReAct loop, Tool Calling).
          \item \textbf{Vai trò của LLM:} Tự chọn gọi tool nào, khi nào dừng lại và rẽ nhánh về đâu.
          \item \textbf{Routing:} Hoàn toàn động dựa trên quan sát (observations) ở runtime.
          \item \textbf{Đánh đổi lớn:} Linh hoạt với bài toán mở, nhưng dễ tích lũy sai số và bùng nổ chi phí.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Định Nghĩa Chuẩn Xác Của Anthropic (2024)}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textit{"Workflows are systems where LLMs and tools are orchestrated through predefined code paths. Agents, on the other hand, are systems where LLMs dynamically direct their own processes and tool usage."}
  \end{tcolorbox}
\end{frame}

% SLIDE 6: VÌ SAO ANTHROPIC KHUYÊN BỎ AGENT
\begin{frame}{Vì Sao Anthropic Khuyến Cáo Ưu Tiên Workflow Thay Vì Agent?}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Hiện tượng lỗi dây chuyền (Cascading Errors), bùng nổ token và rủi ro vận hành}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Cảnh Báo Lỗi Dây Chuyền (Anthropic 2024)}}
        \scriptsize
        \textit{"Each additional step introduces potential failure modes... errors compound."}
        \begin{itemize}\tiny
          \item Trong mô hình chuỗi nối tiếp (\textit{Serial Reliability}), nếu mỗi bước độc lập đạt $p \in [0.90, 0.95]$:
          \item 3 bước có kiểm định: $P_{\text{success}} \approx 0.95^3 \approx \mathbf{85.7\%}$
          \item Vòng lặp tự do 15 turns: $P_{\text{success}} \approx 0.95^{15} \approx \mathbf{46.3\%}$
        \end{itemize}
      \end{block}

      \vspace{-0.1cm}
      \begin{center}
        \begin{tikzpicture}[node distance=0.55cm, auto,
          wf/.style={rectangle, draw=darkgreen, fill=green!8, font=\tiny\bfseries, rounded corners, minimum height=0.45cm},
          ag/.style={rectangle, draw=darkred, fill=red!8, font=\tiny\bfseries, rounded corners, minimum height=0.45cm},
          gate/.style={rectangle, draw=alertorange, fill=orange!15, font=\tiny, rounded corners, minimum height=0.45cm},
          arr/.style={-latex, thick}]

          % Workflow path
          \node[wf] (w1) {Step 1};
          \node[gate, right=0.3cm of w1] (g1) {Check Gate};
          \node[wf, right=0.3cm of g1] (w2) {Step 2};
          \node[right=0.2cm of w2, font=\tiny\color{darkgreen}\bfseries] {Lỗi bị chặn};
          \draw[arr, darkgreen] (w1) -- (g1);
          \draw[arr, darkgreen] (g1) -- (w2);

          % Agent Loop path
          \node[ag, below=0.5cm of w1] (a1) {Turn 1 (Sai)};
          \node[ag, right=0.4cm of a1] (a2) {Turn 2 (Nhiễu)};
          \node[ag, right=0.4cm of a2] (a3) {Turn 3 (Ảo giác)};
          \draw[arr, darkred] (a1) -- (a2);
          \draw[arr, darkred] (a2) -- (a3);
        \end{tikzpicture}
      \end{center}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{alertblock}{\textbf{Bốn Rủi Ro Vận Hành Khi Lạm Dụng Agent Quá Sớm}}
        \scriptsize
        \begin{enumerate}\tiny
          \item \textbf{Độ trễ không giới hạn (Unbounded Latency):} Số turns do LLM tự quyết; thời gian phản hồi biến thiên khôn lường.
          \item \textbf{Chi phí Token bùng nổ:} Lịch sử hội thoại tích lũy dồn toa gây quadratic prompt bloat.
          \item \textbf{Bế tắc khi Debug:} Hành vi non-deterministic khiến việc tái hiện edge-case bugs cực kỳ khó khăn.
          \item \textbf{Trôi lệch ngữ cảnh (Context Drift):} Phản hồi tool rác làm bẩn bộ nhớ, gây ra ảo giác dây chuyền.
        \end{enumerate}
      \end{alertblock}
      \vspace{-0.05cm}
      \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Luận Điểm Cốt Lõi}},fonttitle=\tiny\bfseries]
        \tiny \textbf{Vì sao Anthropic khuyên bắt đầu bằng workflow?} Với 90\% bài toán thực tế, workflow tin cậy hơn, rẻ hơn 3--8 lần và kiểm soát được lỗi tốt hơn hẳn autonomous agent.
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

    \textbf{3. Parallelization} & Fixed Code Path & $1.2\times--2.5\times$ & Thấp nhất (Song song) & Nút thắt tại bước tổng hợp gộp & Bỏ phiếu đa góc nhìn, kiểm tra an toàn Guardrails. \\
    \addlinespace
    \textbf{4. Evaluator-Optimizer} & Cyclic Workflow & $2.0\times--4.0\times$ & Trung bình $\to$ Cao & Bẫy nịnh bợ (sycophancy), vòng lặp suy giảm & Sinh mã nguồn, soạn thảo văn bản học thuật/kỹ thuật. \\
    \addlinespace
    \textbf{5. Orchestrator-Workers} & Phân cấp (Hierarchy) & $\mathbf{3.0\times--8.0\times}$ & Chờ nhánh chậm nhất ($\max t_i$) & Mất mát thông tin khi tổng hợp & \textbf{Nghiên cứu sâu, phân rã đa mục tiêu, cô lập ngữ cảnh.} \\
    \addlinespace
    \textbf{6. Autonomous Agent} & Do Model tự quyết định & $5.0\times--15\times+$ & Không đoán trước được & Bẫy lỗi dây chuyền (Cascading failures) & Thám hiểm không gian mở, giải quyết sự cố sandbox. \\
    \bottomrule
  \end{tabularx}

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Triết Lý Simplicity-First Của Anthropic (2024)}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textbf{Luận điểm cốt lõi:} \textit{"When building applications with LLMs, we recommend finding the simplest solution possible, and only increasing complexity when needed."} Bắt đầu với mẫu thiết kế đơn giản nhất giải quyết được bài toán; chỉ nâng cấp lên Orchestrator hoặc Agent khi có đo đạc thực nghiệm (Evals) chứng minh sự cần thiết.
  \end{tcolorbox}

% SLIDE 16: DYNAMIC ORCHESTRATION & SYSTEMS ARCHITECTURE
\begin{frame}{Dynamic Orchestration: Kiến Trúc Phân Cấp Lead/Subagent}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Phân rã tác vụ động, mô hình điều phối và nguyên tắc cô lập ngữ cảnh}

  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{block}{\textbf{Kiến Trúc Điều Phối Phân Cấp (Lead/Subagent)}}
        \footnotesize
        \begin{enumerate}\scriptsize
          \item \textbf{Lead Orchestrator (Dynamic Planner):}
                Phân tích truy vấn mở ở runtime; tự động phân rã thành các subgoals độc lập (vốn điều lệ vs. tỷ lệ sở hữu nước ngoài).
          \item \textbf{Isolated Subagent Workers:}
                Khởi tạo các ephemeral worker chuyên biệt (Retrieval, Audit, Drafting) thực thi song song trong context boundary riêng biệt.
          \item \textbf{Typed State Handoff Protocols:}
                Giao tiếp liên tác tử qua strict typed schemas (Pydantic), loại bỏ hoàn toàn nhiễu ngôn ngữ tự nhiên.
          \item \textbf{Context Purging Policy:}
                Toàn bộ raw scratchpad và search trace nội bộ bị tiêu hủy ngay sau khi trích xuất kết quả cấu trúc về cho Lead.
          \item \textbf{Centralized State Checkpointer:}
                Lưu trữ mọi chuyển đổi trạng thái xuống bộ nhớ bền bỉ (SQLite/Postgres), bảo đảm tính xác định và khả năng phục hồi.
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.46\textwidth}
      \begin{alertblock}{\textbf{Cơ Sở Kỹ Thuật Của Cô Lập Ngữ Cảnh (Context Isolation)}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Triệt tiêu bùng nổ token bậc hai:} Thay vì nhồi monolithic context khổng lồ ($>30\text{k}$ tokens), mỗi worker chạy trong context sạch $<1,500$ tokens $\to$ Giảm $70-80\%$ attention noise.
          \item \textbf{Chống thoái hóa chú ý (Lost in the Middle):} Tập trung attention weights tối đa vào đúng nhiệm vụ chuyên môn của từng subagent.
          \item \textbf{Khoanh vùng lỗi (Fault Containment):} Kết quả rác hoặc văn bản hết hiệu lực ở retrieval node bị cô lập hoàn toàn, không lây nhiễm vào drafter.
        \end{itemize}
      \end{alertblock}
      \vspace{0.08cm}
      \begin{block}{\scriptsize \textbf{Minh Họa Bằng Kiến Trúc Hệ Thống G4 (\texttt{seminar-s1-7/demo/})}}
        \tiny Triển khai thực tế trong nhóm: \texttt{LegalOrchestrator} điều phối StateGraph DAG (\texttt{demo/web/}) với typed schema Pydantic (\texttt{LegalAgentState}), phân rã song song \texttt{LawGraphAgent} (Retrieval) và \texttt{ClaimAuditorAgent} (Audit Oracle) trước khi bàn giao cho \texttt{DrafterAgent}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 17: AGENT DESIGN PATTERNS: ARCHITECTURAL SELECTION & PRUNING
\begin{frame}{Lựa Chọn \& Cắt Tỉa (Pruning) Kiến Trúc Agent}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Vì sao vòng lặp ReAct tự do thất bại trong các bài toán yêu cầu độ tin cậy cao}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Mẫu Kiến Trúc Bị Loại Bỏ: Vòng Lặp ReAct Tự Do}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Bản chất:} Một LLM duy nhất tự do gọi tools theo chu trình lặp Thought $\to$ Action $\to$ Observation.
          \item \textbf{3 Điểm nghẽn lỗi hệ thống trong Production:}
                \begin{enumerate}\scriptsize
                  \item \textbf{Sai số tích lũy cấp số nhân ($P = \prod p_i$):} Với $p = 0.92$, sau 6 bước $P \approx 60\%$, sau 12 bước $P \approx 36\%$ $\to$ Quá mỏng manh cho các bài toán quan trọng.
                  \item \textbf{Bùng nổ chi phí và độ trễ bậc hai:} Lịch sử hội thoại dài thêm sau mỗi turn khiến tổng token tiêu thụ tăng $O(N^2)$, gây lãng phí lớn.
                  \item \textbf{Zero Fault Containment:} Tiến trình nằm trên RAM volatile; lỗi mạng hoặc timeout ở turn cuối làm sụp đổ toàn bộ pipeline từ đầu.
                \end{enumerate}
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Mẫu Kiến Trúc Được Chọn: Structured Orchestrator-Workers}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Bản chất:} Phân rã bài toán thành đồ thị trạng thái có hướng với quyền tự trị bị chặn biên (Bounded Autonomy).
          \item \textbf{3 Lợi thế kiến trúc vượt trội:}
                \begin{enumerate}\scriptsize
                  \item \textbf{Khoanh vùng sai số:} Worker chỉ có thẩm quyền trong phạm vi hẹp; lỗi của một tác tử bị chặn đứng tại boundary, không lan truyền.
                  \item \textbf{Thực thi bền bỉ (Durable Execution):} Tích hợp checkpointing lưu trữ trạng thái sau từng bước $\to$ Cho phép phục hồi hoặc retry cục bộ.
                  \item \textbf{Kiểm soát ngân sách token \& thời gian:} Áp đặt ngân sách token và số lượt suy luận (max turns) cứng trên từng node, loại bỏ vòng lặp vô tận.
                \end{enumerate}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Bài Học Quyết Định Kiến Trúc Trong Thực Tế}},fonttitle=\footnotesize\bfseries]
    \footnotesize Quyền tự trị không ràng buộc (unconstrained autonomy) là nguồn gốc của các lỗi bất định trong doanh nghiệp. Thiết kế chuẩn mực là kết hợp \textbf{khả năng phân rã linh hoạt của LLM với luồng kiểm soát trạng thái tiền định bằng mã nguồn}.
  \end{tcolorbox}
\end{frame}

% SLIDE 18: FAULT-TOLERANT ARCHITECTURE: DETERMINISTIC GROUND-TRUTH ORACLES
\begin{frame}{Thiết Kế Hệ Thống Chịu Lỗi: Deterministic Oracles vs. LLM Self-Critique}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Tách biệt tầng suy luận khỏi tầng kiểm định bằng các công cụ tiền định độc lập}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{\textbf{Hạn Chế Khi Để LLM Tự Đánh Giá (LLM-as-a-Judge)}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Vòng lặp phân phối tham số (Parametric Circularity):} Yêu cầu LLM "tự rà soát" (\textit{"Are you sure?"}) chỉ tái lấy mẫu từ cùng một phân phối $P_\theta(y \vert x)$ đã sinh lỗi ban đầu $\to$ Không thể phát hiện ảo giác bằng chính mô hình tạo ra ảo giác.
          \item \textbf{Bẫy nịnh bợ (Sycophancy Trap):} Evaluator LLM có xu hướng thiên vị và đồng thuận với candidate output của Generator, đặc biệt khi chia sẻ cùng base model.
          \item \textbf{Điểm mù tri thức thời gian thực:} LLM không thể tự biết một văn bản quy phạm pháp luật đã bị bãi bỏ vào ngày hôm qua nếu không truy vấn live external registry.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Cơ Chế Kiểm Định Bằng Deterministic Oracles}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Tách rời hai mặt phẳng (Decoupled Planes):} Tách biệt tầng suy luận xác suất (probabilistic reasoning) khỏi tầng kiểm định tiền định (deterministic validation).
          \item \textbf{Cổng kiểm tra nhị phân (Binary Assertion Gates):} Sử dụng các công cụ xác định (Linters, Type Checkers, Schema Validators, Database Registry APIs) trả về kết quả nhị phân: \texttt{VALID} hoặc \texttt{INVALID}.
          \item \textbf{Ngắt mạch cứng (Hard Circuit Breakers):} Khi Oracle phát hiện vi phạm, đồ thị lập tức ngắt luồng và kích hoạt nhánh phục hồi có điều kiện.
          \item \textbf{Nhật ký kiểm toán bất biến (Audit Trail):} Lưu trữ bằng chứng nhị phân vào durable state phục vụ giải trình kỹ thuật và pháp lý.
          \item \textbf{Thực chứng trong demo nhóm (\texttt{demo/web/}):} Hàm tiền định \texttt{vbpl\_verify\_oracle} lập tức chặn đứng Nghị định 101/2012 bị bãi bỏ, tránh hoàn toàn bẫy nịnh bợ của LLM.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{So Sánh Các Mô Hình Điều Phối: Centralized, Swarms \& Cyclic Graphs}

        \footnotesize

        \begin{itemize}\scriptsize
          \item Lead điều phối trung tâm; subagents hoạt động độc lập và báo cáo về Lead.

          \item \textbf{Đánh đổi:} Nút thắt độ trễ tại Lead (\textbf{Straggler Effect}: $\max t_i$); chi phí token tổng hợp cao ($3-8\times$).

    \begin{column}{0.32\textwidth}

        \footnotesize
        \textbf{Mạng ngang hàng (Peer-to-Peer):}
        \begin{itemize}\scriptsize
          \item Tác tử tự do handoff và giao tiếp ngang hàng trực tiếp (e.g. OpenAI Swarm).
          \item \textbf{Ưu điểm:} Tính linh hoạt rất cao trong đàm phán đa bên không cấu trúc.
          \item \textbf{Rủi ro chí mạng:} Bùng nổ tin nhắn $O(N^2)$; dễ kẹt vòng lặp ping-pong vô tận; không thể rollback trạng thái bền bỉ.

      \end{alertblock}

    \begin{column}{0.32\textwidth}

        \footnotesize
        \textbf{Kiến trúc LangGraph:}
        \begin{itemize}\scriptsize
          \item Điều phối qua đồ thị trạng thái với strict typed schema và conditional edges.
          \item \textbf{Ưu điểm:} Kiểm soát chặt chẽ chu trình lặp; hỗ trợ durable checkpointing và deterministic time-travel.
          \item \textbf{Đánh đổi:} Đòi hỏi định nghĩa state schema chặt chẽ từ đầu; tính tự do thấp hơn swarm.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
