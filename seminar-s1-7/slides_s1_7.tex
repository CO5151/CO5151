% SLIDE 5: WORKFLOWS VS AUTONOMOUS AGENTS
\begin{frame}{Workflows vs. Autonomous Agents: Who Holds the Wheel?}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ Fixed Code Paths vs. Model-Directed Control Flow}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Workflows (Deterministic)}}
        \begin{itemize}\footnotesize
          \item \textbf{Control}: Code directs execution flow
          \item \textbf{LLM Role}: Scoped task execution at fixed steps
          \item \textbf{Properties}: Deterministic, unit-testable, low variance
          \item \textbf{Primitives}: Prompt chains, routers, parallel map
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Autonomous Agents (Dynamic)}}
        \begin{itemize}\footnotesize
          \item \textbf{Control}: Model dictates tool calls and stopping criteria
          \item \textbf{LLM Role}: Combined planner, executor, and controller
          \item \textbf{Properties}: High flexibility, non-deterministic errors
          \item \textbf{Primitives}: ReAct loops, conversational swarms
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{block}{\textbf{Phong's Discussion Point}}
    \footnotesize Every increment of delegated model autonomy directly trades off production reliability.
  \end{block}
\end{frame}

% SLIDE 6: WHY ANTHROPIC RECOMMENDS AGAINST AGENTS
\begin{frame}{Why Anthropic Tells You: "Don't Build an Agent If You Don't Have To"}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ The Compounding Error Trap \& When to Upgrade}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \textbf{\color{darkred}Three Traps in Autonomous Swarms}:
      \begin{enumerate}\footnotesize
        \item \textbf{Compounding Error ($P = p^n$)}:\\
              $p = 90\% \implies (0.90)^6 \approx 53\%$ (coin-flip reliability)
        \item \textbf{Token \& Latency Inflation}:\\
              Full-history accumulation scales costs at $O(n^2)$
        \item \textbf{Runaway Loops}:\\
              Models repeat failed tool calls without hard exits
      \end{enumerate}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{When to Upgrade to an Agent?}},fonttitle=\footnotesize\bfseries]
        \footnotesize
        \textit{Anthropic's criteria for adding autonomy:}
        \begin{itemize}\footnotesize
          \item Static code branches cannot cover task diversity
          \item Multi-step open-ended search is mandatory
          \item Reliable external verifier exists to prune bad paths
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{block}{\textbf{Phong's Takeaway}}
    \footnotesize Starting with complex agents is premature optimization and self-inflicted technical debt.
  \end{block}
\end{frame}

% SLIDE 7: CORE PATTERNS: CHAINING & ROUTING
\begin{frame}{The Simplest Primitives: Chains and Routers}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ What Prompt Chaining and Routing Buy and Cost}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Prompt Chaining (Sequential)}}
        \begin{itemize}\footnotesize
          \item \textbf{Pattern}: Step A $\to$ Step B $\to$ Step C
          \item \textbf{Advantage}: Modularity; step-level unit testing
          \item \textbf{Cost}: Additive latency; cascade error on Step 1
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Semantic Routing (Classifier)}}
        \begin{itemize}\footnotesize
          \item \textbf{Pattern}: Classify intent $\to$ dispatch specialist prompt
          \item \textbf{Advantage}: Route easy queries to cheap models (Cost $\downarrow$)
          \item \textbf{Cost}: Router is a single point of failure
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{alertblock}{\textbf{Phong's Production Insight}}
    \footnotesize A fast semantic router delivers $10\times$ higher accuracy-per-dollar than open-ended agent debates.
  \end{alertblock}
\end{frame}

% SLIDE 8: WHY TAKING A MAJORITY VOTE IS NOT SELF-CORRECTION
\begin{frame}{Why Taking a Majority Vote Is Not the Same as Self-Correction}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ Understanding Self-Consistency (Wang et al., 2022)}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Self-Consistency (Wang et al., 2022)}}
        \begin{itemize}\footnotesize
          \item \textbf{Mechanism}: Sample $K$ paths at $T>0 \to$ Majority vote
          \item \textbf{Target}: Marginalizes stochastic token decoding noise
          \item \textbf{Limit}: Cannot fix false premises or missing facts
          \item \textbf{Verdict}: Statistical sampling, \textbf{not} self-correction
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Sequential Refinement (Self-Critique)}}
        \begin{itemize}\footnotesize
          \item \textbf{Mechanism}: Output 1 $\to$ Model critique $\to$ Output 2
          \item \textbf{Fatal Flaw}: Re-queries the exact same distribution
          \item \textbf{Pathology}: Model rationalizes errors and hallucinated laws
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Phong's Defense Argument}},fonttitle=\footnotesize\bfseries]
    \footnotesize Sampling independent paths filters noise; prompting an LLM to critique itself without tools amplifies bias.
  \end{tcolorbox}
\end{frame}

% SLIDE 9: WHEN TO STICK WITH SIMPLE CODE
\begin{frame}{Decision Framework: When to Stick with Simple Code}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ Three Practical Checks Before Adding Agents}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Stick with Code (90\% of Tasks)}}
        \begin{itemize}\footnotesize
          \item Execution path is predictable ahead of time
          \item Zero tolerance for hallucination or drift
          \item Tight latency and predictable token budgets
          \item Steps can be verified by deterministic unit tests
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Consider an Agent (Only When...)}}
        \begin{itemize}\footnotesize
          \item Execution path cannot be pre-enumerated
          \item Problem requires multi-step hypothesis search
          \item Reliable external checker exists (compiler / API)
          \item Application can afford a $3\times$--$8\times$ token multiplier
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Phong's Engineering Verdict}},fonttitle=\footnotesize\bfseries]
    \footnotesize If you cannot write a reliable automated test to verify the agent's output, giving it autonomy only automates failure.
  \end{tcolorbox}
\end{frame}

% SLIDE 10: WHY NOT JUST STUFF EVERYTHING INTO ONE GIANT CONTEXT WINDOW?
\begin{frame}{Why Not Just Stuff Everything into One Giant Context Window?}
  \framesubtitle{Presenter: Vu Viet Hung $\vert$ Answering the Teacher's Mandatory Question}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Approach A: "One Huge Context" (1M+ Tokens)}}
        \begin{itemize}\footnotesize
          \item \textbf{Lost in the Middle}: Attention degrades on internal facts
          \item \textbf{Context Pollution}: Tool errors contaminate entire prompt
          \item \textbf{Sequential Latency}: Cannot parallelize sub-retrievals
          \item \textbf{Debug Nightmare}: Impossible to isolate error origin
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Approach B: Orchestrator-Workers}}
        \begin{itemize}\footnotesize
          \item \textbf{Context Isolation}: Workers receive only scoped parameters
          \item \textbf{Parallel Fan-Out}: Concurrent worker execution speeds latency
          \item \textbf{Specialized Prompts}: Compliance Drafter $\neq$ Claim Auditor
          \item \textbf{Unit Testable}: Workers independently verifiable \& mockable
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Hung's Core Point on the "Orchestration Tax"}},fonttitle=\footnotesize\bfseries]
    \footnotesize Orchestrators incur a token tax for coordination, but buy \textbf{context isolation and parallelism} that mega-prompts cannot match.
  \end{tcolorbox}
\end{frame}

% SLIDE 11: THE LEAD/SUBAGENT PATTERN: BIG GAINS, BUT AT 8X THE BILL
\begin{frame}{The Lead/Subagent Pattern: Big Gains, but at 8x the Bill}
  \framesubtitle{Presenter: Vu Viet Hung $\vert$ Anthropic's Multi-Agent Research System (2025)}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Evaluator-Optimizer Loop}}
        \begin{itemize}\footnotesize
          \item Worker drafts $\to$ Evaluator grades $\to$ Re-draft loop
          \item Terminates on threshold criteria or max retries
          \item \textbf{Hidden Trap}: Same model weights share identical blind spots
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Lead/Subagents (Anthropic 2025)}}
        \begin{itemize}\footnotesize
          \item \textbf{Lead Agent}: Decomposes master plan, synthesizes output
          \item \textbf{Subagents}: Independent search threads; short summaries
          \item \textbf{The Trade-Off}: Unlocks multi-hour tasks, but incurs a **$3\times$--$8\times$ token multiplier**
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{block}{\textbf{Hung's Defense Talking Point}}
    \footnotesize Anthropic rejected uncoordinated subagent chats because peer-to-peer swarms converge on mutual agreement over errors.
  \end{block}
\end{frame}

% SLIDE 12: WHAT HAPPENS WHEN YOUR AGENT CRASHES 45 MINUTES IN?
\begin{frame}{What Happens When Your Agent Crashes 45 Minutes In?}
  \framesubtitle{Presenter: Vu Viet Hung $\vert$ LangGraph, Durable Execution \& Checkpointing}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \textbf{\color{hcmutblue}Production Reality: Long Runs Break}
      \begin{itemize}\footnotesize
        \item Network timeouts, rate limits, and server crashes
        \item Without persistence: State lost $\to$ re-run from scratch (cost $\uparrow$)
      \end{itemize}
      \vspace{0.1cm}
      \textbf{\color{hcmutblue}LangGraph Architecture}:
      \begin{itemize}\footnotesize
        \item \textbf{StateGraph}: Explicit state schema with pure reducers
        \item \textbf{Checkpointers}: Snapshots saved to SQLite/Postgres per node
        \item \textbf{Human Gate}: Pause execution for approval; resume state
      \end{itemize}
    \end{column}

    \begin{column}{0.47\textwidth}
      \begin{tcolorbox}[colback=boxbg,colframe=darkgreen,title={\textbf{Time-Travel & Fault Recovery}},fonttitle=\footnotesize\bfseries]
        \begin{itemize}\footnotesize
          \item \textbf{Rewind}: Rollback to step $N-1$ on error
          \item \textbf{Resume}: Reboot continues from last verified node
          \item \textbf{Audit Trace}: Full state inspectability for compliance
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{alertblock}{\textbf{Hung's Takeaway}}
    \footnotesize Checkpointing turns fragile toy scripts into durable production software.
  \end{alertblock}
\end{frame}

% SLIDE 13: WHY ASKING AN LLM TO "DOUBLE-CHECK" MAKES IT WORSE
\begin{frame}{Why Asking an LLM to "Double-Check" Usually Makes Things Worse}
  \framesubtitle{Presenter: Vu Viet Hung $\vert$ Huang et al. (2023): The Myth of Intrinsic Self-Correction}

  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \textbf{\color{darkred}Huang et al. (ICLR 2024) Proof}:
      \begin{itemize}\footnotesize
        \item LLMs \textbf{cannot self-correct reasoning} without external signals
        \item Under prompt: \textit{"Review and fix your mistakes"}:
          \begin{itemize}\footnotesize
            \item Wrong $\to$ Right: $< 5\%$ of benchmark cases
            \item Right $\to$ Wrong: $15\% - 25\%$ (overthinking degradation)
          \end{itemize}
        \item \textbf{Conclusion}: Intrinsic self-critique degrades net accuracy
      \end{itemize}
    \end{column}

    \begin{column}{0.45\textwidth}
      \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Why Self-Correction Fails}},fonttitle=\footnotesize\bfseries]
        \begin{enumerate}\footnotesize
          \item \textbf{Shared Blind Spot}: Same weights cannot detect own logical premises
          \item \textbf{Sycophantic Doubt}: "Are you sure?" prompts panic reversals
        \end{enumerate}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{alertblock}{\textbf{Hung's Provocation for the Committee}}
    \footnotesize Early self-refine papers showed gains only in prose writing, or accidentally leaked answers in critique prompts.
  \end{alertblock}
\end{frame}

      \textbf{Design Pattern} & \textbf{Who Controls Flow} & \textbf{Token Cost} & \textbf{Latency} & \textbf{What It Buys You} & \textbf{What It Costs You} \\
      \midrule
      \textbf{Prompt Chaining} & Code (Fixed) & $1\times$ & Linear & Step modularity; clear unit tests & Added latency; cascade failure \\
      \textbf{Routing} & Code + Classifier & $1-1.2\times$ & Lowest & Route by cost and prompt specialization & Misrouting ruins the whole run \\
      \textbf{Parallelization} & Code (Fan-out) & $N\times$ & Fast & Fast throughput; diverse perspectives & Multiplies tokens; aggregator bottleneck \\
      \textbf{Orchestrator-Workers} & Lead LLM & $3-5\times$ & Moderate & Context isolation; dynamic subgoals & Orchestration tax; lost cross-context \\
      \textbf{Evaluator-Optimizer} & Feedback Loop & $2-6\times$ & High & Iterative improvement against criteria & Risk of infinite loops; shared blind spots \\
      \textbf{Autonomous Agent} & Unconstrained LLM & $5-15\times$ & Very High & Open tool exploration; zero-shot plans & Non-deterministic; runaway errors \\
      \bottomrule
    \end{tabularx}
  \end{table}

  \vspace{0.15cm}
  \begin{block}{\footnotesize \textbf{Hung's Rule of Thumb}}
    \footnotesize Start with \textbf{Chaining + Routing}. Move to \textbf{Orchestrator-Workers} only when the problem requires dynamic task planning. Avoid open-ended swarms in production.
  \end{block}
\end{frame}

% SLIDE 17: CATCHING MISTAKES EARLY (PRM VS ORM)
\begin{frame}{Catching Mistakes Early: Why Step-by-Step Verification Beats Final Scores}
  \framesubtitle{Presenter: Dang Lam Tung (Lead) $\vert$ Lightman et al. (2023 - OpenAI): Let's Verify Step by Step}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Outcome Models (ORM): Final Score}}
        \begin{itemize}\footnotesize
          \item Evaluates only final answer ($0$ or $1$)
          \item \textbf{Fatal Flaw}: Rewards broken logic that got lucky
          \item Cannot assign credit to intermediate steps
          \item Ineffective for tree search (MCTS)
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Process Models (PRM): Step-by-Step}}
        \begin{itemize}\footnotesize
          \item Evaluates validity of \textbf{each individual reasoning step}
          \item Identifies the exact inflection point of error
          \item Enables early pruning of bad search branches
          \item Powers test-time compute scaling (e.g. OpenAI o1/o3)
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Significance for Agent Architecture}},fonttitle=\footnotesize\bfseries]
    \footnotesize Process supervision converts agents from unguided text generators into step-verified tree search systems.
  \end{tcolorbox}
\end{frame}

% SLIDE 18: TEACHING A MODEL BY KEEPING WHAT WORKS (STAR)
\begin{frame}{Teaching a Model to Reason by Keeping Only What It Gets Right}
  \framesubtitle{Presenter: Dang Lam Tung (Lead) $\vert$ STaR (Zelikman et al., 2022): Bootstrapping Reasoning}

  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{The STaR Bootstrap Loop}}
        \begin{enumerate}\footnotesize
          \item \textbf{Generate}: Sample step-by-step reasoning paths
          \item \textbf{Filter (The Gate)}: Check against \textbf{Ground-Truth Answer Key}\\
                $\to$ \textit{Discard 100\% of incorrect trajectories}
          \item \textbf{Rationalize}: If failed, provide correct label $\to$ reconstruct
          \item \textbf{Finetune}: Train model weights only on verified paths
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.47\textwidth}
      \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Signal Origin}},fonttitle=\footnotesize\bfseries]
        \begin{itemize}\footnotesize
          \item STaR does not self-learn in a vacuum
          \item 100\% of improvement comes from the \textbf{Answer Key}
          \item Removing external ground truth triggers training collapse
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{block}{\textbf{Tung's Insight}}
    \footnotesize Self-training without an external ground-truth filter is mathematically impossible.
  \end{block}
\end{frame}
