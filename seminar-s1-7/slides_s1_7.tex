% SLIDE 5: WORKFLOWS VS AUTONOMOUS AGENTS
\begin{frame}{Workflows vs. Autonomous Agents: Who Holds the Wheel?}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Fixed Code Paths vs. Model-Directed Control Flow}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Workflows: Fixed Code Paths}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Control Flow:} Hard-coded bằng code truyền thống (Python, Go).
          \item \textbf{Role of LLM:} Atomic reasoning step bên trong các pipeline stages định sẵn.
          \item \textbf{Routing:} Logic if-else, deterministic schema validation, deterministic state transitions.
          \item \textbf{Key Advantages:} Highly predictable, reproducible, latency thấp, token overhead tối thiểu ($1.0\times$).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Autonomous Agents: Model-Directed Control}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Control Flow:} Do vòng lặp suy luận của LLM dẫn dắt (ReAct loop, Tool Calling).
          \item \textbf{Role of LLM:} Mô hình tự quyết định gọi tool nào, khi nào dừng lại và rẽ nhánh về đâu.
          \item \textbf{Routing:} Động hoàn toàn dựa trên quan sát (observations) ở runtime.
          \item \textbf{Key Trade-off:} Linh hoạt cao với bài toán mở, nhưng dễ tích lũy sai số và bùng nổ chi phí ngoài kiểm soát.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Anthropic's Exact Definition (2024)}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textit{"Workflows are systems where LLMs and tools are orchestrated through predefined code paths. Agents, on the other hand, are systems where LLMs dynamically direct their own processes and tool usage."}
  \end{tcolorbox}
\end{frame}

% SLIDE 6: VÌ SAO ANTHROPIC KHUYÊN BỎ AGENT
\begin{frame}{The Defense Question: Why Anthropic Recommends Against Agents}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Compounding Error ($P = p^n$), Cost Inflation, và Debugging Friction}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{The Compounding Error Trap ($P = p^n$)}}
        \footnotesize
        Giả sử mỗi bước gọi tool mô hình đạt độ chính xác $p \in [0.90, 0.95]$:
        \begin{itemize}\scriptsize
          \item \textbf{3-step workflow:} $P_{\text{success}} = 0.95^3 \approx \mathbf{85.7\%}$
          \item \textbf{8-step autonomous loop:} $P_{\text{success}} = 0.95^8 \approx \mathbf{66.3\%}$
          \item \textbf{15-step agent loop:} $P_{\text{success}} = 0.95^{15} \approx \mathbf{46.3\%}$
        \end{itemize}
        \vspace{0.1cm}
        \alert{\textbf{Key Finding:}} Trong unconstrained agent loop, độ tin cậy của toàn hệ thống giảm theo exponential ($P = p^n$) khi số bước thực thi (trajectory steps) tăng.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{The Four Hidden Costs of Premature Autonomy}}
        \footnotesize
        \begin{enumerate}
          \item \textbf{Unbounded Latency:} Số turns do LLM quyết định; thời gian phản hồi biến thiên khôn lường.
          \item \textbf{Runaway Token Costs:} Lịch sử hội thoại tích lũy liên tục gây quadratic prompt bloat.
          \item \textbf{Debugging Nightmare:} Hành vi non-deterministic khiến việc tái hiện edge-case bugs cực kỳ khó khăn.
          \item \textbf{Context Drift:} Các phản hồi tool rác làm bẩn context memory, gây ra hallucination cascades.
        \end{enumerate}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Direct Defense for Q\&A}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textbf{Why does Anthropic recommend against agents for many tasks?} Because for 90\% of production applications, workflows provide higher reliability, lower latency, 3--8x lower cost, and simpler debugging than autonomous agents.
  \end{tcolorbox}
\end{frame}

% SLIDE 7: PROMPT CHAINING & ROUTING
\begin{frame}{Concrete Patterns 1 \& 2: Prompt Chaining and Routing}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Foundational Workflow Primitives: What Each Buys and Costs}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Pattern 1: Prompt Chaining}}
        \footnotesize
        \textbf{Cơ chế:} Chia nhỏ bài toán phức tạp thành chuỗi các lệnh gọi LLM tuần tự; bước sau tiêu thụ đầu ra của bước trước.\\
        \vspace{0.1cm}
        \textbf{What It Buys:}
        \begin{itemize}\scriptsize
          \item Giảm tải cognitive load trên mỗi prompt.
          \item Cho phép chèn programmatic validation giữa các bước.
          \item Cho phép dùng mô hình nhỏ, rẻ hơn cho intermediate steps.
        \end{itemize}
        \textbf{What It Costs:}
        \begin{itemize}\scriptsize
          \item Linear latency penalty ($T = \sum t_i$).
          \item Inflexible: Không thể tự phục hồi nếu bước trước sinh ra lỗi ngoài dự tính mà không có explicit branching code.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Pattern 2: Routing}}
        \footnotesize
        \textbf{Cơ chế:} Một classifier (LLM hoặc regex/embedding) nhận diện input intent để dẫn hướng tới specialized downstream path.\\
        \vspace{0.1cm}
        \textbf{What It Buys:}
        \begin{itemize}\scriptsize
          \item Domain specialization: Mỗi nhánh có tailored system prompt và narrow tools.
          \item Ngăn ngừa prompt bloat (tránh nhồi 50 tools vào 1 prompt).
          \item Tối ưu chi phí: Route các truy vấn đơn giản sang small models.
        \end{itemize}
        \textbf{What It Costs:}
        \begin{itemize}\scriptsize
          \item Router misclassification sẽ làm hỏng toàn bộ pipeline con.
          \item Maintenance overhead: Thêm danh mục mới đòi hỏi cập nhật router và downstream handlers.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 8: EVALUATOR-OPTIMIZER PATTERN
\begin{frame}{Concrete Pattern 3: Evaluator-Optimizer Feedback Loops}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Iterative Refinement: What It Buys, What It Costs, and Failure Modes}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{Evaluator-Optimizer Mechanism}}
        \footnotesize
        \textbf{Hai vai trò tách biệt rõ rệt:}
        \begin{enumerate}\scriptsize
          \item \textbf{Generator LLM:} Tạo ra candidate output dựa trên yêu cầu.
          \item \textbf{Evaluator (LLM hoặc Tool):} Đánh giá candidate dựa trên explicit criteria và trả về structured feedback.
          \item \textbf{Refinement Loop:} Generator điều chỉnh lại output dựa trên feedback cho đến khi vượt qua threshold.
        \end{enumerate}
        \vspace{0.1cm}
        \textbf{What It Buys:}
        \begin{itemize}\scriptsize
          \item Cải thiện output quality rõ rệt trong code generation, translation, và formal drafting.
          \item Mô phỏng quy trình drafting và peer-review của con người.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Critical Costs \& Failure Modes}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Token Explosion:} $K$ iterations làm tăng token consumption lên $K \times$, độ trễ kéo dài tương ứng.
          \item \textbf{Sycophantic Agreement Trap:} Nếu Evaluator là LLM cùng base weights, nó thường nịnh bợ Generator và chấp thuận hallucinations tinh vi.
          \item \textbf{Diminishing Returns:} Chất lượng tăng mạnh nhất ở loop 1; các loops sau dễ bị circular regression.
          \item \textbf{Remedy:} Bắt buộc tích hợp \textbf{external deterministic tools} (linters, test suites, API checkers) thay vì pure LLM self-reflection.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 13: LANGGRAPH STATE GRAPHS
\begin{frame}{LangGraph: Multi-Agent Workflows as Cyclic State Graphs}
  \framesubtitle{Thuyết trình: Vũ Việt Hưng $\vert$ Extension Reading 3: State Graphs, Reducers, and Conditional Routing}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{LangGraph Core Concepts}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Centralized State Schema:} Shared, typed state dictionary được truyền xuyên suốt qua tất cả các nodes.
          \item \textbf{State Reducers:} Functions quy định cách node updates merge vào global state (e.g., append messages vs. overwrite).
          \item \textbf{Nodes as Execution Units:} Mỗi node là một LLM call, một isolated subagent, hoặc một deterministic tool.
          \item \textbf{Conditional Edges:} Python functions kiểm tra state để route execution dynamically (nếu audit fail $\to$ route to recovery).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Why Cyclic State Graphs Matter}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Traditional DAGs are strictly forward-only:} Nếu retrieval lấy phải revoked statute, DAG không thể backtrack.
          \item \textbf{Cyclic State Graphs allow loops:} Graph có thể route back về planning node để tìm văn bản thay thế kèm retry counter.
          \item \textbf{Bounded Loops:} Graph edges enforce max retry limits (FSM circuit breaker) để triệt tiêu hoàn toàn infinite loops.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Engineering Value}},fonttitle=\footnotesize\bfseries]
    \footnotesize LangGraph bridges the gap between rigid deterministic workflows and unconstrained agents by providing \textbf{structured, inspectable state transitions}.
  \end{tcolorbox}
\end{frame}

          \item \textbf{Human Review:} Human expert inspects state, chỉnh sửa dữ liệu nếu cần, và approve resumption.
          \item \textbf{Time-Travel Debugging:} Developers có thể fork execution từ bất kỳ historical checkpoint nào để test prompt variants hoặc trace errors.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{2. Isolate Contexts}}
        \footnotesize
        \begin{itemize}
          \item Khi bắt buộc dùng multi-agent, adopt Anthropic's Lead/Subagent split.
          \item Never share a giant messy scratchpad across agents.
          \item Chấp nhận token multiplier $3-8\times$ như cái giá của context hygiene.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{3. Build for Durability}}
        \footnotesize
        \begin{itemize}
          \item Always implement state checkpointing (LangGraph style).
          \item Anchor agent actions to external ground-truth oracles.
          \item Design explicit re-route edges với bounded retry limits.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.25cm}
  \begin{alertblock}{\textbf{The Golden Engineering Rule}}
    \footnotesize \textbf{Core Takeaway:} The best agent architecture is the simplest one that solves the task reliably; complexity is an engineering cost, not a feature.
  \end{alertblock}
