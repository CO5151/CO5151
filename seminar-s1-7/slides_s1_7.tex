\begin{frame}{1. Bối Cảnh: Vì Sao Single-Prompt \& Naive Chain Thất Bại?}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \textbf{\color{darkred}Ba Giới Hạn Cốt Tử Của LLM Đơn Tuyến:}
      \begin{itemize}
        \item \textbf{No Self-Correction (Không tự sửa):} Mô hình tự sinh token tiếp theo dựa trên xác suất; sai lầm ở bước 1 sẽ thành tiên đề tuyệt đối cho bước 2.
        \item \textbf{Context Degradation (Suy thoái ngữ cảnh):} Đưa toàn bộ tài liệu và công cụ vào một prompt làm loãng chú ý (Lost in the Middle).
        \item \textbf{Combinatorial Complexity:} Bài toán thế giới thực đòi hỏi phân rã đa tầng (Sub-task decomposition), gọi công cụ ngoài và xử lý phi tất định.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\color{hcmutblue}Sự Chuyển Dịch Mô Hình (Paradigm Shift)}
        \centering
        \small
        \textbf{Single Prompt} ($y = f(x)$)\\
        $\Downarrow$\\
        \textbf{Static Chain / Naive RAG} ($x \to R \to y$)\\
        $\Downarrow$\\
        \textbf{Autonomous Agent Orchestration}\\
        $\mathcal{S}_{t+1}, \mathcal{A}_t = \pi(\mathcal{S}_t, \mathcal{H}_t, \mathcal{E}_t)$
      \end{block}
      \vspace{0.1cm}
      {\footnotesize \textit{"Chất lượng của một hệ thống Agentic không nằm ở trọng số mô hình lớn tới đâu, mà nằm ở kiến trúc điều phối luồng trạng thái."} -- Anthropic, 2024.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{2. Định Nghĩa Hình Thức Bài Toán Điều Phối Tác Tử (Orchestration)}
  \begin{block}{Hệ Thống Điều Phối Đa Tác Tử Được Định Nghĩa Bởi Bộ 5:}
    $$\mathcal{M} = \langle \mathcal{S}, \mathcal{A}, \mathcal{T}, \mathcal{O}, \mathcal{C} \rangle$$
  \end{block}
  \begin{itemize}
    \item $\mathcal{S}$ -- Không gian trạng thái hệ thống: $\mathcal{S}_t = \langle \text{Memory}, \text{Active Sub-goals}, \text{Artifacts}, \text{Error Budget} \rangle$.
    \item $\mathcal{A}$ -- Không gian hành động: Gồm suy luận nội tại (Thought), gọi công cụ (Tool Call), hoặc ủy quyền tác vụ (Delegate to Worker).
    \item $\mathcal{T}: \mathcal{S} \times \mathcal{A} \to \mathcal{S}$ -- Hàm chuyển trạng thái có xác thực (State Transition Function).
    \item $\mathcal{O}$ -- Bộ điều phối (Orchestrator Policy): Quyết định rẽ nhánh, dừng hoặc gọi lại phản hồi.
    \item $\mathcal{C}$ -- Chi phí tổng hợp: $\text{Cost} = \alpha \cdot \text{Token} + \beta \cdot \text{Latency} + \gamma \cdot \text{ErrorRate}$.
  \end{itemize}
  \vspace{0.2cm}
  \textbf{\color{darkred}Mục tiêu cốt lõi:} Cực tiểu hóa $\mathcal{C}$ trong khi vẫn đảm bảo nghiệm hội tụ và chính xác hoàn toàn.
\end{frame}

\begin{frame}{4. Pattern 1: Prompt Chaining \& Semantic Routing}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\color{hcmutblue}Nguyên lý Hoạt Động}
        \begin{itemize}
          \item Phân rã bài toán lớn thành các bước xác định liên tiếp $Step_1 \to Step_2 \to \dots \to Step_k$.
          \item Sử dụng bộ phân loại ngữ nghĩa (Semantic Router / Embeddings) để hướng luồng sang prompt chuyên biệt.
        \end{itemize}
      \end{block}
      \textbf{\color{darkgreen}Ưu điểm:}
      \begin{itemize}
        \item Cực kỳ ổn định (Deterministic), chi phí token $O(N)$ thấp nhất.
        \item Dễ quan sát (Observability cao), dễ viết unit test từng khâu.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\color{darkred}Điểm gãy cốt tử (Failure Boundary):}
      \begin{itemize}
        \item \textbf{Cascading Error:} Bước 1 trích xuất sai thực thể $\to$ các bước sau suy luận trên dữ liệu rác.
        \item \textbf{Rigid Topology:} Hoàn toàn bất lực nếu bài toán có cấu trúc rẽ nhánh không dự đoán trước được.
      \end{itemize}
      \begin{exampleblock}{Khi nào nên dùng?}
        Quy trình xử lý hồ sơ chuẩn có mẫu sẵn, trích xuất hóa đơn, pipeline ETL dữ liệu.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{5. Pattern 2: Parallelization (Sectioning vs Voting)}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{A. Sectioning (Chia nhỏ tác vụ song song):}
      \begin{itemize}
        \item Chia input thành nhiều phân đoạn độc lập, gọi nhiều LLMs xử lý cùng lúc và tổng hợp lại.
        \item Ví dụ: Tóm tắt 5 chương sách độc lập; trích xuất rủi ro từ 10 hợp đồng khác nhau.
        \item Giảm thiểu Wall-clock Latency từ $O(K \cdot t) \to O(t)$.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{B. Voting (Biểu quyết đa số / Self-Consistency):}
      \begin{itemize}
        \item Chạy cùng 1 bài toán trên $N$ phiên bản với nhiệt độ $T > 0$ hoặc trên các mô hình khác nhau.
        \item Dùng Majority Vote hoặc Judge Agent để chọn kết quả đồng thuận cao nhất.
        \item Tăng mạnh độ chính xác trong suy luận logic và toán học.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{alertblock}{Đánh Đổi Kỹ Thuật (Trade-off)}
    Tiết kiệm thời gian người dùng nhưng nhân tuyến tính chi phí API (\$ Token Cost tăng gấp $K$ lần).
  \end{alertblock}
\end{frame}

\begin{frame}{6. Pattern 3: Orchestrator-Workers (Điều Phối Động)}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{Orchestrator:} Đọc yêu cầu phức tạp $\to$ phân tích ngữ cảnh $\to$ lập kế hoạch chia nhỏ thành các nhiệm vụ con (Sub-tasks).
        \item \textbf{Workers:} Các tác tử chuyên môn hóa sâu (chỉ cầm một số công cụ nhất định) thực thi nhiệm vụ được giao.
        \item \textbf{Synthesizer:} Gom kết quả từ các Workers và tổng hợp câu trả lời cuối cùng.
      \end{itemize}
    \end{column}

\begin{frame}{7. Pattern 4: Evaluator-Optimizer (Vòng Lặp Phản Biện)}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{\color{hcmutblue}Cấu Trúc Tác Tử Kép (Generator vs Critic)}
        \begin{enumerate}
          \item \textbf{Generator:} Khởi tạo lời giải / bản thảo đầu tiên.
          \item \textbf{Evaluator:} Đánh giá lời giải dựa trên tiêu chí đo lường định lượng hoặc rubric khắt khe.
          \item \textbf{Looping Condition:} Nếu chưa đạt ngưỡng $\to$ cung cấp phản hồi phản biện (Feedback) để Generator viết lại.
        \end{enumerate}
      \end{block}
      \textbf{Các nghiên cứu nền tảng:} *Reflexion* (Shinn et al., 2023), *Self-Refine* (Madaan et al., 2023), *CRITIC* (Gou et al., 2024).
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{darkred}Hai Căn Bệnh Của Evaluator-Optimizer:}
      \begin{itemize}
        \item \textbf{Self-Confirmation Bias:} Nếu dùng cùng một model cho cả Generator và Evaluator trong cùng context window, Evaluator luôn có xu hướng khen bài của chính mình.
        \item \textbf{Infinite Regress / Oscillation:} Hai tác tử tranh cãi qua lại quanh các chi tiết phong cách mà không hội tụ về bản chất logic.
      \end{itemize}
      \begin{alertblock}{Giải Pháp G4 Đề Xuất}
        Phải cô lập hoàn toàn context window và bắt buộc Evaluator phải truy xuất Ground-Truth Tools ngoài.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{8. Pattern 5: Multi-Agent Systems (Hierarchical vs Swarm)}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{\color{hcmutblue}A. Phân Tầng Chỉ Huy (Hierarchical - MetaGPT)}
      \begin{itemize}
        \item Mô phỏng cơ cấu tổ chức doanh nghiệp: CEO $\to$ Architect $\to$ Engineer $\to$ QA.
        \item Chuẩn hóa quy trình thao tác chuẩn (SOPs) để định hướng thông điệp.
        \item Giảm thiểu tính hỗn loạn nhưng triệt tiêu tính thích nghi trước ngoại lệ.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{alertorange}B. Mạng Lưới Tự Do (Swarm / AutoGen)}
      \begin{itemize}
        \item Các tác tử trò chuyện tự do qua Group Chat hoặc bảng tin chung (Blackboard pattern).
        \item Khả năng giải quyết vấn đề sáng tạo cao trong các bài toán mở.
        \item \textbf{Hậu quả:} Tăng bùng nổ token giao tiếp (Communication Tax), dễ rơi vào buồng vang (Echo Chamber).
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{center}
    \textit{"Giao tiếp tự do giữa các agent thường làm tăng độ nhiễu nhiều hơn là tăng tri thức."}
  \end{center}
\end{frame}

\begin{frame}{9. Quản Lý Trạng Thái \& Phân Tầng Bộ Nhớ (State \& Memory)}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{Ephemeral Scratchpad (Ngắn hạn):} Lưu trữ dấu vết suy luận tức thời (ReAct Thought/Action/Observation) trong cửa sổ ngữ cảnh.
        \item \textbf{Session State (Đồ thị trạng thái hữu hạn - FSM):} Lưu vết qua schema có định kiểu (Typed Schema trong LangGraph). Đảm bảo tính kiểm soát (Checkpoints, Replay, Time-travel).
        \item \textbf{Long-term Memory (Dài hạn):} Lưu trữ kiến thức và kinh nghiệm qua Vector Store và Knowledge Graph.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\color{hcmutblue}Nguyên Tắc Bất Biến Về Quản Lý Trạng Thái}
        \begin{itemize}
          \item \textbf{State Redundancy:} Tránh nhồi toàn bộ lịch sử trò chuyện vào prompt; chỉ lưu state biến thiên.
          \item \textbf{Deterministic Guardrails:} Các chuyển dịch trạng thái then chốt phải được kiểm soát bằng code truyền thống, không phó mặc hoàn toàn cho LLM.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{12. Phân Tích Phản Biện I: Lan Truyền Lỗi \& Suy Giảm Mục Tiêu}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\color{darkred}Toán Học Của Sự Lan Truyền Sai Số}
        Xét một chuỗi tác tử gồm $K$ bước suy luận độc lập, mỗi bước có xác suất chính xác $p$:
        $$P(\text{Toàn hệ thống đúng}) = \prod_{i=1}^K p_i = p^K$$
        \begin{itemize}
          \item Nếu $p = 0.90$ (mô hình rất tốt) và $K = 6$ bước:
          $$P(\text{Thành công}) = 0.90^6 \approx \mathbf{53.1\%}$$
          \item Một nửa số yêu cầu sẽ thất bại dù từng bước riêng lẻ trông rất hoàn hảo!
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{darkred}Hiện Tượng "Goal Drift" (Lệch Khỏi Mục Tiêu):}
      \begin{itemize}
        \item Khi thông điệp đi qua 3 tầng giao tiếp (Orchestrator $\to$ Lead Worker $\to$ Sub-Worker), các ràng buộc phủ định (Negative Constraints) bị rơi rụng.
        \item Tác tử thực thi sau cùng thường tối ưu hóa cục bộ một mục tiêu không còn liên quan đến yêu cầu gốc của người dùng.
      \end{itemize}
      \begin{exampleblock}{Biện Pháp Đối Phó}
        Cơ chế \textbf{Goal Invariant Checkpoint} đối chiếu trực tiếp với State gốc sau mỗi bước.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{13. Phân Tích Phản Biện II: Những Giả Định Ẩn Thường Bị Phớt Lờ}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Giả Định 1: "LLM Phân Rã Bài Toán Hoàn Hảo"}
        \textbf{Thực tế:} Mô hình phân rã dựa trên tương đồng ngữ nghĩa bề mặt, thường xuyên bỏ sót các tiền điều kiện pháp lý/kỹ thuật tiềm ẩn (Implicit Dependencies).
      \end{alertblock}
      \begin{alertblock}{Giả Định 2: "Tool Luôn Trả Output Chuẩn Xác"}
        \textbf{Thực tế:} Thế giới thực đầy rẫy lỗi timeout, API rate limit, HTML rác, schema thay đổi đột ngột. Tác tử không biết cách xử lý lỗi âm thầm (Silent Truncation).
      \end{alertblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Giả Định 3: "Multi-Agent Luôn Tốt Hơn Single-Agent"}
        \textbf{Thực tế:} Giao tiếp đa tác tử làm tăng chi phí token lên gấp 10--30 lần. Nhiều nghiên cứu (Chen et al., 2024) chứng minh phần lớn hiệu năng tăng thêm chỉ đến từ việc nạp thêm compute (Best-of-N sampling) chứ không phải do sự tương tác thông minh.
      \end{alertblock}
      \begin{alertblock}{Giả Định 4: "Evaluator Luôn Khách Quan"}
        \textbf{Thực tế:} LLM Evaluator bị chi phối bởi độ dài câu trả lời (Verbosity bias) và tính cả nể (Sycophancy).
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{14. What we don't believe / what the papers don't show}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\color{darkred}1. Ảo Tưởng Về "Tính Trồi" (Emergence)}
        \textbf{Báo chí ca ngợi:} Sự hợp tác đa tác tử tự phát sinh trí tuệ tập thể.\\
        \textbf{What we don't believe:} Không hề có phép màu nổi trội. Chuẩn hóa ngân sách token, Single-Agent chạy Sampling Best-of-N đánh bại 80\% các kiến trúc Multi-Agent phức tạp.
      \end{block}
      \begin{block}{\color{darkred}2. Bệnh Tự Nịnh Bợ (Sycophancy Trap)}
        \textbf{Các paper trình diễn:} Các Agent tranh luận sôi nổi và cải thiện kết quả.\\
        \textbf{What papers don't show:} Nếu không có công cụ kiểm chứng khách quan (Ground Truth), Agent Auditor luôn nhượng bộ và đồng thuận mù quáng với Drafter sau 2--3 lượt chat.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{\color{darkred}3. Con Số Token \& Độ Trễ Bị Che Giấu}
        \textbf{Các paper công bố:} Tăng Pass@1 từ 65\% lên 85\%.\\
        \textbf{What papers don't show:} Chi phí token tăng 2.400\% và P99 latency vọt lên 50 giây. Mô hình không thể thương mại hóa được trong thực tế người dùng.
      \end{block}
      \begin{block}{\color{darkred}4. Benchmark Trong Môi Trường Đóng}
        \textbf{Các bài báo benchmark:} HumanEval, GSM8K, SWE-bench.\\
        \textbf{What we don't believe:} Các môi trường này có sẵn Unit Test rõ ràng. Khi đưa vào môi trường mở phi đơn điệu (như Pháp luật Việt Nam), không có unit test nào, Agent Evaluator lập tức mất phương hướng.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{20. Kết Luận: Thông Điệp Cốt Lõi Về Agent Orchestration}
  \begin{enumerate}
    \item \textbf{No Silver Bullet:} Không có một Pattern nào là vạn năng. Chọn đúng Pattern là bài toán tối ưu đa mục tiêu giữa \textbf{Độ trễ}, \textbf{Chi phí} và \textbf{Độ tin cậy}.
    \item \textbf{Simplicity Over Complexity:} Luôn bắt đầu từ Simple Prompt Chaining $\to$ Routing $\to$ Evaluator-Optimizer. Chỉ chuyển sang Multi-Agent khi có lý do kỹ thuật bắt buộc phải phân tách vai trò.
    \item \textbf{Grounding Trumps Debate:} Tranh luận giữa các LLM không tạo ra chân lý. Hệ thống chỉ đáng tin cậy khi được neo vào các công cụ xác thực thế giới thực (Ground-truth Verification).
    \item \textbf{State Machines Win in Production:} Các đồ thị trạng thái hữu hạn có định kiểu (FSM / LangGraph) là tiêu chuẩn vàng cho các ứng dụng công nghiệp, thay thế hoàn toàn cho các phòng chat tự do vô định.
  \end{enumerate}
\end{frame}
