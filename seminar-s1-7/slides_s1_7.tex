% SLIDE 5: WORKFLOWS VS AUTONOMOUS AGENTS
\begin{frame}{Workflows vs. Agents: Ai Nắm Control Flow?}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Code path cố định hay để model tự quyết định}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Workflows: Control Flow Nằm Trong Code}}
        \scriptsize
        \begin{itemize}\scriptsize
          \item \textbf{Control flow:} viết sẵn bằng code (Python, Go...).
          \item \textbf{Vai trò của LLM:} thực hiện từng bước nhỏ trong một pipeline đã định trước.
          \item \textbf{Rẽ nhánh:} if-else, kiểm tra schema, chuyển trạng thái theo quy tắc cố định.
          \item \textbf{Ưu điểm:} dễ dự đoán, dễ tái lập, latency thấp, ít token ($1\times$).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Agents: Model Tự Quyết Định Control Flow}}
        \scriptsize
        \begin{itemize}\scriptsize
          \item \textbf{Control flow:} do LLM quyết định trong vòng lặp (ReAct, tool calling).
          \item \textbf{Vai trò của LLM:} tự chọn tool, quyết định khi nào dừng và đi tiếp hướng nào.
          \item \textbf{Rẽ nhánh:} thay đổi theo observation ở runtime.
          \item \textbf{Đánh đổi:} linh hoạt với bài toán mở, nhưng lỗi dễ cộng dồn và chi phí khó kiểm soát.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.1cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[height=1.65cm]{figures/anthropic_augmented_llm.png}\par
      \tiny \textbf{Building block chung:} augmented LLM = LLM + retrieval, tools, memory.
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[height=1.65cm]{figures/anthropic_autonomous_agent.png}\par
      \tiny \textbf{Agent:} LLM gọi tool trong vòng lặp, nhận feedback từ environment và tự quyết khi nào dừng.
    \end{column}
  \end{columns}
  \vspace{0.05cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Định Nghĩa Của Anthropic (2024)}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textit{"Workflows are systems where LLMs and tools are orchestrated through predefined code paths. Agents, on the other hand, are systems where LLMs dynamically direct their own processes and tool usage, maintaining control over how they accomplish tasks."}
  \end{tcolorbox}
\end{frame}

% SLIDE 6: VÌ SAO ANTHROPIC KHUYÊN ƯU TIÊN WORKFLOW
\begin{frame}{Vì Sao Anthropic Khuyên Ưu Tiên Workflow?}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Lỗi cộng dồn ($P = p^n$), chi phí token và rủi ro vận hành}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{Lỗi Cộng Dồn Theo Số Bước ($P = p^n$)}}
        \scriptsize
        Giả sử mỗi bước đúng với xác suất $p = 0.95$ và các bước độc lập với nhau:
        \begin{itemize}\scriptsize
          \item \textbf{Workflow 3 bước:} $P_{\text{success}} = 0.95^3 \approx \mathbf{85.7\%}$
          \item \textbf{Agent loop 8 bước:} $P_{\text{success}} = 0.95^8 \approx \mathbf{66.3\%}$
          \item \textbf{Agent loop 15 bước:} $P_{\text{success}} = 0.95^{15} \approx \mathbf{46.3\%}$
        \end{itemize}
        \vspace{0.1cm}
        \alert{\textbf{Nhận xét:}} Độ tin cậy giảm theo hàm mũ khi trajectory dài ra. Workflow biết trước số bước và chèn được validation giữa các bước; agent thì không biết trước mình sẽ chạy bao nhiêu bước.
      \end{block}

      \begin{center}
        \begin{tikzpicture}[
          wf/.style={rectangle, draw=darkgreen, fill=green!8, font=\tiny\bfseries, rounded corners, minimum height=0.45cm},
          ag/.style={rectangle, draw=darkred, fill=red!8, font=\tiny\bfseries, rounded corners, minimum height=0.45cm},
          gate/.style={rectangle, draw=alertorange, fill=orange!15, font=\tiny, rounded corners, minimum height=0.45cm},
          arr/.style={-latex, thick}]
          \node[wf] (w1) {Step 1};
          \node[gate, right=0.3cm of w1] (g1) {Check Gate};
          \node[wf, right=0.3cm of g1] (w2) {Step 2};
          \node[right=0.2cm of w2, font=\tiny\color{darkgreen}\bfseries] {Lỗi bị chặn lại};
          \draw[arr, darkgreen] (w1) -- (g1);
          \draw[arr, darkgreen] (g1) -- (w2);
          \node[ag, below=0.3cm of w1.south west, anchor=north west] (a1) {Turn 1: sai};
          \node[ag, right=0.4cm of a1] (a2) {Turn 2: nhiễu};
          \node[ag, right=0.4cm of a2] (a3) {Turn 3: hallucination};
          \draw[arr, darkred] (a1) -- (a2);
          \draw[arr, darkred] (a2) -- (a3);
        \end{tikzpicture}
      \end{center}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Bốn Rủi Ro Khi Trao Quyền Tự Chủ Quá Sớm}}
        \scriptsize
        \begin{enumerate}
          \item \textbf{Unbounded latency:} số turn do model quyết định nên thời gian phản hồi khó dự đoán.
          \item \textbf{Token tăng nhanh:} history dồn lên sau mỗi turn, tổng token tăng gần $O(N^2)$.
          \item \textbf{Khó debug:} hành vi non-deterministic, rất khó tái hiện lỗi ở edge case.
          \item \textbf{Context drift:} tool output nhiễu tích tụ trong context, kéo theo hallucination dây chuyền.
        \end{enumerate}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Trả Lời Câu Hỏi Trong Đề Cương}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textbf{Vì sao Anthropic khuyên không dùng agent cho phần lớn bài toán?} Vì \textit{"agentic systems often trade latency and cost for better task performance"}. Với task đã rõ các bước, workflow ổn định hơn, rẻ hơn và dễ debug hơn. Anthropic viết thẳng: \textit{"This might mean not building agentic systems at all."}
  \end{tcolorbox}
\end{frame}

% SLIDE 7: PROMPT CHAINING & ROUTING
\begin{frame}{Pattern Cơ Bản: Prompt Chaining và Routing}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Cách hoạt động, lợi ích, cái giá và use case}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Pattern 1: Prompt Chaining}}
        \centering
        \includegraphics[width=0.88\linewidth]{figures/anthropic_prompt_chaining.png}\par
        \raggedright\scriptsize
        \textbf{Cơ chế:} chuỗi LLM call nối tiếp, gate bằng code kiểm tra ở giữa.\\
        \textbf{What it buys:}
        \begin{itemize}\scriptsize
          \item Mỗi prompt chỉ lo một việc, dễ viết và dễ kiểm soát.
          \item Gate chặn lỗi sớm; bước trung gian dùng được model nhỏ.
        \end{itemize}
        \textbf{What it costs:}
        \begin{itemize}\scriptsize
          \item Latency cộng dồn theo số bước ($T = \sum t_i$).
          \item Không có nhánh rẽ thì lỗi trôi thẳng xuống bước sau.
        \end{itemize}
      \end{block}
      \begin{tcolorbox}[usecase,title={Use case}]
        \scriptsize
        \begin{itemize}\scriptsize
          \item Viết marketing copy, rồi dịch sang ngôn ngữ khác.
          \item Viết outline $\rightarrow$ kiểm tra outline đạt tiêu chí $\rightarrow$ viết tài liệu theo outline.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Pattern 2: Routing}}
        \centering
        \includegraphics[height=1.1cm]{figures/anthropic_routing.png}\par
        \raggedright\scriptsize
        \textbf{Cơ chế:} classifier (LLM hoặc model truyền thống) phân loại input, chuyển sang nhánh chuyên biệt.\\
        \textbf{What it buys:}
        \begin{itemize}\scriptsize
          \item Mỗi nhánh có system prompt và bộ tool riêng, tránh prompt bloat.
          \item Tối ưu cho loại input này không làm hỏng loại input khác.
        \end{itemize}
        \textbf{What it costs:}
        \begin{itemize}\scriptsize
          \item Router phân loại sai thì cả nhánh phía sau sai theo.
          \item Thêm nhánh mới phải sửa cả router lẫn handler.
        \end{itemize}
      \end{block}
      \begin{tcolorbox}[usecase,title={Use case}]
        \scriptsize
        \begin{itemize}\scriptsize
          \item Customer service: general question, refund request và technical support được chuyển vào ba nhánh riêng.
          \item Câu hỏi dễ sang Claude Haiku, câu hỏi khó sang Claude Sonnet để tối ưu chi phí.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

        \includegraphics[width=0.85\linewidth]{figures/anthropic_parallelization.png}\par
        \raggedright\scriptsize
        Nhiều LLM call chạy đồng thời, \textbf{code} gộp kết quả (aggregator). Hai dạng:
        \begin{itemize}\scriptsize
          \item \textbf{Sectioning:} chia task thành các phần độc lập, mỗi phần một LLM call.
          \item \textbf{Voting:} chạy cùng một task nhiều lần để có nhiều output rồi gộp lại.
        \end{itemize}
        \textbf{What it buys:}
        \begin{itemize}\scriptsize
          \item Nhanh hơn khi các phần không phụ thuộc nhau.
          \item Mỗi call chỉ tập trung vào một khía cạnh nên làm tốt hơn; voting cho kết quả đáng tin hơn.
        \end{itemize}
        \textbf{What it costs:}
        \begin{itemize}\scriptsize
          \item $N$ nhánh thì tốn khoảng $N\times$ token; latency bằng nhánh chậm nhất ($\max t_i$).
          \item Phải viết logic gộp: chọn ngưỡng vote, xử lý khi các nhánh mâu thuẫn.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[usecase,title={Use case}]
        \scriptsize
        \textbf{Sectioning:}
        \begin{itemize}\scriptsize
          \item \textbf{Guardrails:} một instance trả lời user, một instance khác song song lọc nội dung không phù hợp; hiệu quả hơn so với dồn cả hai việc vào một call.
          \item \textbf{Automated evals:} mỗi LLM call chấm một khía cạnh khác nhau của output.
        \end{itemize}
        \textbf{Voting:}
        \begin{itemize}\scriptsize
          \item \textbf{Review code tìm vulnerability:} nhiều prompt khác nhau cùng review, prompt nào thấy vấn đề thì flag.
          \item \textbf{Kiểm duyệt nội dung:} nhiều prompt đánh giá các khía cạnh khác nhau, dùng ngưỡng vote để cân bằng false positive và false negative.
        \end{itemize}
      \end{tcolorbox}
      \begin{alertblock}{\textbf{Khác Gì Orchestrator-Workers?}}
        \scriptsize Hình dạng giống nhau, nhưng ở parallelization các subtask được \textbf{định nghĩa sẵn trong code}. Ở orchestrator-workers, subtask do \textbf{orchestrator LLM quyết định} theo từng input (slide 11).
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 9: EVALUATOR-OPTIMIZER PATTERN
\begin{frame}{Pattern Có Feedback: Evaluator-Optimizer}
  \framesubtitle{Thuyết trình: Nguyễn Trung Phong $\vert$ Vòng lặp tinh chỉnh, use case và bẫy sycophancy}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{Cơ Chế Evaluator-Optimizer}}
        \centering
        \includegraphics[width=0.95\linewidth]{figures/anthropic_evaluator_optimizer.png}\par
        \raggedright\scriptsize
        \textbf{Hai vai trò tách riêng:}
        \begin{enumerate}\scriptsize
          \item \textbf{Generator:} tạo solution từ yêu cầu.
          \item \textbf{Evaluator (LLM hoặc tool):} chấm theo tiêu chí rõ ràng; \textit{Rejected} thì trả feedback, \textit{Accepted} thì ra output.
          \item \textbf{Refinement loop:} generator sửa theo feedback cho tới khi đạt.
        \end{enumerate}
        \textbf{Dấu hiệu phù hợp (Anthropic):} khi có feedback của con người, output tốt lên rõ rệt; \textit{và} LLM cũng tự đưa ra được feedback tương tự.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Chi Phí \& Các Failure Mode Thường Gặp}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Token tăng theo số vòng:} $K$ vòng thì tốn khoảng $K\times$ token, latency cũng tăng theo.
          \item \textbf{Sycophancy:} nếu evaluator dùng chung base model với generator, nó dễ ``gật đầu'' và bỏ sót những lỗi khó thấy.
          \item \textbf{Diminishing returns:} chất lượng thường tăng nhiều nhất ở vòng đầu; các vòng sau dễ lòng vòng mà không tiến bộ.
          \item \textbf{Cách khắc phục:} dùng \textbf{external deterministic tools} (linter, unit test, API checker) làm evaluator thay vì để LLM tự review.
        \end{itemize}
      \end{alertblock}
      \begin{tcolorbox}[usecase,title={Use case}]
        \scriptsize
        \begin{itemize}\scriptsize
          \item \textbf{Literary translation:} bản dịch đầu bỏ sót sắc thái, evaluator chỉ ra chỗ cần sửa.
          \item \textbf{Complex search:} nhiều vòng search và phân tích; evaluator quyết định có cần search tiếp không.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 12: ORCHESTRATOR-WORKERS VS NGỮ CẢNH KHỔNG LỒ
\begin{frame}{Orchestrator-Workers vs. Single Context}
  \framesubtitle{Thuyết trình: Vũ Việt Hưng $\vert$ Context Isolation vs. Attention Dilution trong Agentic Systems}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Hạn Chế Của Single Long Context}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Attention Dilution \& Context Rot:} Nhồi nhét các ngữ cảnh thô (logs, tools, history) làm tụt dốc tỉ lệ Signal-to-Noise. Mô hình dễ bỏ sót ràng buộc và gặp hiện tượng \textit{Lost in the Middle}.
          \item \textbf{Truncation:} Khi chạm trần ngữ cảnh, cơ chế cắt tỉa bắt buộc diễn ra, dễ làm rụng mất system prompt và kế hoạch ban đầu.
          \item \textbf{Compounding Errors:} Lỗi ở một bước gọi công cụ sẽ bị "neo" vào chuỗi tự hồi quy kế tiếp, khiến agent rơi vào vòng lặp vô tận hoặc trôi khỏi mục tiêu.
          \item \textbf{Sequential Execution \& High Cost:} Tiến trình bị khóa trong chuỗi tuần tự $O(N)$; mọi lượt sinh token sau đều chịu chi phí và độ trễ tính toán trên toàn bộ lịch sử thô.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Lợi Thế Của Orchestrator-Workers}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Context Isolation:} Mỗi subagent sở hữu context window riêng và chỉ nhận đúng công cụ chuyên biệt, loại bỏ rủi ro hallucinated tool arguments.
          \item \textbf{Information Compression:} Worker xử lý hàng chục nghìn tokens thô nhưng chỉ bàn giao bản tóm tắt tinh gọn cho Orchestrator, giữ sạch context cấp cao.
          \item \textbf{Parallel Tooling:} Phân rã bài toán mở rộng theo chiều ngang cho nhiều worker chạy đồng thời, cắt giảm độ trễ thực thi lên tới 90\%.
          \item \textbf{Compute Scaling \& Model Tiering:} Cho phép tăng token budget hiệu dụng vượt trần đơn agent và phối hợp linh hoạt mô hình lớn với mô hình nhẹ, rẻ.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

      \begin{block}{\textbf{Durable Execution Là Gì?}}
        \scriptsize
        \begin{itemize}
          \item Agent chạy task dài \textbf{chắc chắn sẽ gặp sự cố}: rớt mạng, dính rate limit, container bị kill.
          \item \textbf{Không có checkpoint:} lỗi ở bước 9 nghĩa là chạy lại từ bước 1, tốn thêm cả token lẫn thời gian.
          \item \textbf{Checkpointer của LangGraph:} lưu state xuống SQLite/Postgres sau mỗi bước của graph.
          \item \textbf{Fault recovery:} chạy tiếp từ checkpoint hợp lệ gần nhất.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Time Travel \& Human-in-the-Loop}}
        \scriptsize
        \begin{itemize}
          \item \textbf{Interrupts:} graph tạm dừng trước các hành động rủi ro cao.
          \item \textbf{Human review:} người kiểm tra state, chỉnh sửa nếu cần rồi cho chạy tiếp.
          \item \textbf{Time travel:} fork từ một checkpoint bất kỳ trong quá khứ để thử lại hoặc truy vết lỗi.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[colback=boxbg,colframe=darkgreen,title={\textbf{Bài Học Theo Đề Cương}},fonttitle=\footnotesize\bfseries]
    \footnotesize \textbf{Durable execution biến một agent prototype dễ vỡ thành hệ thống có thể đưa vào production}, vì phần việc đã làm xong không bị mất khi có sự cố.
  \end{tcolorbox}

% SLIDE 18: DYNAMIC ORCHESTRATION
\begin{frame}{Dynamic Orchestration: Kiến Trúc Lead/Subagent Phân Cấp}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Phân rã task động và context isolation}

  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{block}{\textbf{Kiến Trúc Lead/Subagent Phân Cấp}}
        \scriptsize
        \begin{enumerate}\scriptsize
          \item \textbf{Lead orchestrator (dynamic planner):}
                phân tích query ở runtime, tách thành các subgoal độc lập (ví dụ: vốn điều lệ và tỷ lệ sở hữu nước ngoài).
          \item \textbf{Isolated subagent workers:}
                tạo worker ngắn hạn cho từng việc (retrieval, audit, drafting), chạy song song, mỗi worker có context riêng.
          \item \textbf{Typed state handoff:}
                các agent trao đổi qua schema có kiểu (Pydantic) thay vì văn bản tự do.
          \item \textbf{Context purging:}
                scratchpad và search trace của worker bị bỏ đi sau khi đã trích xuất kết quả có cấu trúc gửi về lead.
          \item \textbf{Centralized checkpointer:}
                lưu mọi lần chuyển state xuống SQLite/Postgres để có thể khôi phục khi cần.
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.46\textwidth}
      \begin{alertblock}{\textbf{Vì Sao Context Isolation Hiệu Quả?}}
        \scriptsize
        \begin{itemize}
          \item \textbf{Context nhỏ và sạch:} thay vì một context khổng lồ chứa mọi thứ, mỗi worker chỉ thấy phần thông tin của mình.
          \item \textbf{Giảm ``lost in the middle'':} thông tin cần dùng không bị chìm giữa hàng chục nghìn token không liên quan.
          \item \textbf{Fault containment:} kết quả rác hay văn bản hết hiệu lực ở node retrieval bị chặn lại, không lan sang drafter.
        \end{itemize}
      \end{alertblock}

      \end{block}

  \end{columns}
\end{frame}

% SLIDE 19: CHỌN & LOẠI PATTERN
\begin{frame}{Chọn \& Loại Pattern: Vì Sao Không Dùng ReAct Tự Do?}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Quyết định kiến trúc cho bài toán cần độ tin cậy cao}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Pattern Bị Loại: ReAct Loop Không Giới Hạn}}
        \scriptsize
        \begin{itemize}
          \item \textbf{Bản chất:} một LLM tự gọi tool theo vòng Thought $\to$ Action $\to$ Observation.
          \item \textbf{3 vấn đề khi chạy production:}
                \begin{enumerate}\scriptsize
                  \item \textbf{Lỗi cộng dồn ($P = \prod p_i$):} với $p = 0.92$, 6 bước còn $P \approx 60\%$, 12 bước còn $P \approx 36\%$.
                  \item \textbf{Token tăng $O(N^2)$:} history dài thêm sau mỗi turn và bị gửi lại toàn bộ.
                  \item \textbf{Không có fault containment:} state chỉ nằm trong RAM; lỗi mạng hay timeout ở turn cuối là mất hết.
                \end{enumerate}
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Pattern Được Chọn: Orchestrator-Workers Có Cấu Trúc}}
        \scriptsize
        \begin{itemize}
          \item \textbf{Bản chất:} bài toán được tổ chức thành state graph; mỗi node chỉ có quyền tự chủ trong giới hạn (bounded autonomy).
          \item \textbf{3 lợi thế:}
                \begin{enumerate}\scriptsize
                  \item \textbf{Fault containment:} worker chỉ có quyền hạn hẹp; lỗi bị chặn tại ranh giới của worker.
                  \item \textbf{Durable execution:} checkpoint sau mỗi bước, có thể retry cục bộ.
                  \item \textbf{Budget control:} giới hạn token và max turns cho từng node, không có vòng lặp vô hạn.
                \end{enumerate}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.05cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Bài Học Rút Ra}},fonttitle=\footnotesize\bfseries]
    \scriptsize Unconstrained autonomy là nguồn gốc chính của những lỗi khó đoán. Thiết kế hợp lý là để LLM lo phần phân rã task linh hoạt, còn \textbf{control flow và state do code quản lý}.
  \end{tcolorbox}
\end{frame}

% SLIDE 20: DETERMINISTIC ORACLES
\begin{frame}{Fault Tolerance: Deterministic Oracles vs. LLM Self-Critique}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Tách phần reasoning khỏi phần validation}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{\textbf{Giới Hạn Của LLM-as-a-Judge}}
        \scriptsize
        \begin{itemize}
          \item \textbf{Parametric circularity:} hỏi lại model \textit{"Are you sure?"} thực chất là lấy mẫu lại từ chính phân phối $P_\theta(y \vert x)$ đã sinh ra lỗi. Không có feedback bên ngoài, self-correction thường không cải thiện được kết quả (Huang et al., 2024).
          \item \textbf{Sycophancy:} evaluator LLM dễ đồng tình với output của generator, nhất là khi dùng chung base model.
          \item \textbf{Không có thông tin real-time:} model không thể tự biết một văn bản luật vừa bị bãi bỏ hôm qua nếu không tra registry bên ngoài.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Validation Bằng Deterministic Oracles}}
        \scriptsize
        \begin{itemize}
          \item \textbf{Decoupled planes:} tách phần reasoning (mang tính xác suất) khỏi phần validation (tiền định).
          \item \textbf{Binary assertion gates:} dùng linter, type checker, schema validator, registry API... trả về kết quả rõ ràng \texttt{VALID} hoặc \texttt{INVALID}.
          \item \textbf{Hard circuit breakers:} khi oracle phát hiện vi phạm, graph dừng nhánh hiện tại và chuyển sang nhánh recovery.
          \item \textbf{Audit trail:} lưu kết quả kiểm tra vào durable state để truy vết về sau, cả về kỹ thuật lẫn pháp lý.
          \item \textbf{Trong demo của nhóm:} hàm tiền định \texttt{external\_vbpl\_verify\_oracle} phát hiện Nghị định 101/2012 đã bị bãi bỏ mà không cần hỏi LLM.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 21: STATE PERSISTENCE & ROLLBACK
\begin{frame}{Xử Lý Lỗi \& Rollback State Trong Graph}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Xử lý lỗi cục bộ và rollback dựa trên checkpoint}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{Luồng Thực Thi \& Differential Rollback}}
        \footnotesize
        \begin{enumerate}\scriptsize
          \item \textbf{Checkpoint $C_1$ (planning):} orchestrator tách query thành Subgoal 1 (điều kiện cấp phép) và Subgoal 2 (tỷ lệ sở hữu nước ngoài) $\to$ lưu $C_1$ (SQLite/Postgres).
          \item \textbf{Chạy song song:} Worker 1 xử lý Subgoal 1, Worker 2 xử lý Subgoal 2 $\to$ lưu $C_2$.
          \item \textbf{Oracle phát hiện lỗi:} auditor thấy văn bản dùng cho Subgoal 1 đã hết hiệu lực $\to$ kích hoạt conditional failure edge.
          \item \textbf{Differential rollback:}
                $\bullet$ Quay state về $C_1$.\\
                $\bullet$ \textbf{Giữ lại kết quả hợp lệ:} Subgoal 2 vẫn nằm trong state, không phải chạy lại.\\
                $\bullet$ Chỉ chạy lại Subgoal 1 với văn bản thay thế, retry tối đa $K \le 2$ lần.
          \item \textbf{Hoàn tất:} drafter tổng hợp hồ sơ từ dữ liệu đã qua kiểm tra $\to$ lưu $C_3$.
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Hệ Thống Đảm Bảo Được Gì?}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Không cold restart:} không chạy lại các subtask đã hợp lệ, tiết kiệm token và thời gian.
          \item \textbf{Resumable:} rớt mạng hay container sập thì chạy tiếp từ checkpoint gần nhất.
          \item \textbf{Time-travel debugging:} fork từ một checkpoint bất kỳ để tái hiện và chẩn đoán lỗi.
          \item \textbf{Xem trực tiếp trong demo:} web demo hiển thị working memory, Pydantic schema và bảng checkpoint SQLite theo thời gian thực.
        \end{itemize}
      \end{alertblock}

        \tiny Trace xử lý việc Nghị định 101/2012 bị bãi bỏ và thay bằng Nghị định 52/2024, chạy trên \texttt{LegalOrchestrator} (\texttt{seminar-s1-7/demo/}, giao diện tại \texttt{localhost:8765}). Checkpoint được lưu bằng \texttt{SQLiteCheckpointManager} do nhóm tự viết theo nguyên lý của LangGraph.
      \end{block}

\end{frame}

% SLIDE 22: ORCHESTRATION TOPOLOGIES
\begin{frame}{So Sánh Các Topology: Hub-and-Spoke, Swarm \& Cyclic Graph}
  \framesubtitle{Thuyết trình: Đặng Lâm Tùng (Trưởng nhóm) $\vert$ Ba cách tổ chức giao tiếp giữa các agent}

  \begin{columns}[t]
    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{1. Hub-and-Spoke (Anthropic)}}
        \footnotesize
        \textbf{Mô hình tập trung:}
        \begin{itemize}\scriptsize
          \item Lead điều phối; subagent làm việc độc lập rồi báo cáo lại cho lead.
          \item \textbf{Ưu điểm:} lead nắm toàn bộ state; context của các subagent tách biệt.
          \item \textbf{Đánh đổi:} latency phụ thuộc subagent chậm nhất (\textbf{straggler effect}, $\max t_i$); tốn nhiều token (khoảng $15\times$ chat).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{alertblock}{\textbf{2. Decentralized Swarm}}
        \footnotesize
        \textbf{Peer-to-peer:}
        \begin{itemize}\scriptsize
          \item Agent tự handoff và trao đổi trực tiếp với nhau (ví dụ: OpenAI Swarm, nay là OpenAI Agents SDK).
          \item \textbf{Ưu điểm:} linh hoạt với hội thoại nhiều bên, không cần cấu trúc cố định.
          \item \textbf{Rủi ro:} số message có thể tăng $O(N^2)$; dễ kẹt vòng lặp ping-pong; khó rollback state.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{3. Cyclic State Graph}}
        \footnotesize
        \textbf{LangGraph:}
        \begin{itemize}\scriptsize
          \item Điều phối qua state graph với typed schema và conditional edges.
          \item \textbf{Ưu điểm:} kiểm soát được vòng lặp; có sẵn checkpointing và time travel.
          \item \textbf{Đánh đổi:} phải thiết kế state schema cẩn thận từ đầu; kém linh hoạt hơn swarm.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Lựa Chọn Cho Hệ Thống Production}},fonttitle=\footnotesize\bfseries]
    \footnotesize Hệ thống cần độ tin cậy cao nên tránh swarm không có ràng buộc. Hướng phổ biến hiện nay là \textbf{hub-and-spoke để phân công việc}, kết hợp với \textbf{cyclic state graph để xử lý lỗi và recovery}.
  \end{tcolorbox}
\end{frame}
