% SLIDE 5: WORKFLOWS VS AUTONOMOUS AGENTS
\begin{frame}{Workflows vs. Autonomous Agents: Who Holds the Wheel?}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ Fixed Code Paths vs. Model-Directed Control Flow}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Workflows: Code Controls the Model}}
        \begin{itemize}\footnotesize
          \item \textbf{Who decides the path}: Software logic written by engineers.
          \item \textbf{What the LLM does}: Handles content at specific designated stops.
          \item \textbf{What you get}: Deterministic behavior, easy unit testing, low variance.
          \item \textbf{Everyday examples}: Step-by-step chains, routers, parallel map-reduce.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Autonomous Agents: The Model Controls the Path}}
        \begin{itemize}\footnotesize
          \item \textbf{Who decides the path}: The LLM decides what tools to call and when to stop.
          \item \textbf{What the LLM does}: Acts as both planner, worker, and controller.
          \item \textbf{What you get}: High flexibility, but high risk of surprises and failure.
          \item \textbf{Everyday examples}: Free-form ReAct loops, conversational swarms.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.25cm}
  \begin{block}{\textbf{Phong's Discussion Question}}
    \footnotesize
    \begin{itemize}
      \item \textbf{The reality check}: If you wrap an LLM in a `while True` loop and let it pick tools, is it an intelligent agent, or just an unpredictable state machine?
      \item \textbf{The trade-off}: Every bit of autonomy you hand over to the model costs you reliability in production.
    \end{itemize}
  \end{block}
\end{frame}

% SLIDE 6: WHY ANTHROPIC RECOMMENDS AGAINST AGENTS
\begin{frame}{Why Anthropic Tells You: "Don't Build an Agent If You Don't Have To"}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ The Compounding Error Trap \& When to Upgrade}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \textbf{\color{darkred}Three Traps That Kill Autonomous Agents}:
      \begin{enumerate}\footnotesize
        \item \textbf{Mistakes Multiply Exponentially}:
              If each step is $90\%$ accurate, after 6 steps:
              $0.90^6 \approx 53\%$ success rate (a coin flip!).
        \item \textbf{Tokens and Latency Balloon}:
              Every turn drags along old context. Costs grow as $O(n^2)$, and response times jump from seconds to minutes.
        \item \textbf{Getting Stuck in Loops}:
              Without hardcoded guardrails, models repeat failed tool calls or drift away from the user's original goal.
      \end{enumerate}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{When Are You Actually Allowed to Build an Agent?}},fonttitle=\footnotesize\bfseries]
        \footnotesize
        \textit{Anthropic's rule: Stay with simple code until:}
        \begin{itemize}
          \item[$\blacksquare$] You cannot write out the possible branches ahead of time.
          \item[$\blacksquare$] The task requires genuinely open multi-step search.
          \item[$\blacksquare$] You have an external checker (a compiler or database) to catch bad trajectories.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{block}{\textbf{Phong's Takeaway}}
    \footnotesize Starting with complex agents isn't sophisticated engineering; it's self-inflicted technical debt.
  \end{block}
\end{frame}

% SLIDE 7: CORE PATTERNS: CHAINING & ROUTING
\begin{frame}{The Simplest Primitives: Chains and Routers}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ What Prompt Chaining and Routing Buy and Cost}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Prompt Chaining (Step-by-Step)}}
        \footnotesize
        \textbf{How it works}: Break a big prompt into Step A $\to$ Step B $\to$ Step C.\\
        \textbf{What you gain}:
        \begin{itemize}\scriptsize
          \item Clean modularity; you can inspect and test each step.
          \item Less cognitive overload per prompt.
        \end{itemize}
        \textbf{What you pay}:
        \begin{itemize}\scriptsize
          \item Latencies add up sequentially.
          \item If Step 1 hallucinates, Step 2 builds on falsehoods.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Semantic Routing (Sorting at the Door)}}
        \footnotesize
        \textbf{How it works}: Inspect the request first, send it to a custom specialist.\\
        \textbf{What you gain}:
        \begin{itemize}\scriptsize
          \item Cost efficiency: Cheap, small models handle easy questions.
          \item High prompt focus: Specialists get clean, uncluttered instructions.
        \end{itemize}
        \textbf{What you pay}:
        \begin{itemize}\scriptsize
          \item The router is a single point of failure.
          \item A wrong turn at the door ruins the whole answer.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{alertblock}{\textbf{Phong's Production Insight}}
    \footnotesize In real systems, a fast semantic router gives you higher accuracy per dollar than any complex multi-agent framework.
  \end{alertblock}
\end{frame}

% SLIDE 8: WHY TAKING A MAJORITY VOTE IS NOT SELF-CORRECTION
\begin{frame}{Why Taking a Majority Vote Is Not the Same as Self-Correction}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ Understanding Self-Consistency (Wang et al., 2022)}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Self-Consistency: Voting Across Samples}}
        \footnotesize
        \textbf{How it works}:
        \begin{enumerate}\scriptsize
          \item Ask the model the same question $K$ times independently.
          \item Let each run reason freely at a slightly warm temperature.
          \item Pick whichever final answer appears most often.
        \end{enumerate}
        \textbf{Why it works}: It washes out random decoding mistakes.\\
        \textbf{The key fact}: \textbf{No self-correction is happening here!} The samples never read each other or fix each other.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Sequential Refinement: Editing Old Output}}
        \footnotesize
        \textbf{How it works}:
        \begin{enumerate}\scriptsize
          \item Model writes Answer 1.
          \item Model reads Answer 1 and writes a critique.
          \item Model rewrites Answer 1 into Answer 2.
        \end{enumerate}
        \textbf{The fatal flaw}:
        \begin{itemize}\scriptsize
          \item Errors chain together.
          \item If the model didn't know the answer in Step 1, its critique in Step 2 is just as confused.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Phong's Defense Argument}},fonttitle=\footnotesize\bfseries]
    \footnotesize Voting filters out random noise across independent attempts. Self-refinement assumes the model can accurately judge its own errors — an assumption that regularly fails.
  \end{tcolorbox}
\end{frame}

% SLIDE 9: WHY BORING CODE WINS: REJECTED SWARMS IN LEGAL AI
\begin{frame}{Why Boring Code Wins: Why G4 Rejected Swarms in Legal AI}
  \framesubtitle{Presenter: Nguyen Trung Phong $\vert$ Group G4 Production Case Study in Vietnamese Law}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{What We Rejected: Conversational Swarms}}
        \footnotesize
        \textit{Multiple agents chatting freely (AutoGen style)}:
        \begin{itemize}\scriptsize
          \item Let a "Drafter" and "Auditor" debate legal decrees.
          \item \textbf{Why we threw it out}:
            \begin{itemize}\tiny
              \item Zero tolerance for hallucination in statutory law.
              \item The agents flattered each other and agreed on revoked laws.
              \item Token costs exploded while accuracy dropped.
            \end{itemize}
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{What We Built: Deterministic StateGraph}}
        \footnotesize
        \textit{Strict code graph with external verification}:
        \begin{itemize}\scriptsize
          \item Router sorts civil, traffic, or corporate questions.
          \item Knowledge graph enforces legal hierarchy (Law $\succ$ Decree $\succ$ Circular).
          \item \textbf{Hard reality check}: External government API confirms whether a decree is still in effect before the model can quote it.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Phong's Engineering Verdict}},fonttitle=\footnotesize\bfseries]
    \footnotesize When legal compliance or money is on the line, \textbf{code must keep the control}. Never leave document validity to model consensus.
  \end{tcolorbox}
\end{frame}

% SLIDE 10: WHY NOT JUST STUFF EVERYTHING INTO ONE GIANT CONTEXT WINDOW?
\begin{frame}{Why Not Just Stuff Everything into One Giant Context Window?}
  \framesubtitle{Presenter: Vu Viet Hung $\vert$ Answering the Teacher's Mandatory Question}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Idea A: "One Huge Context" (1M+ Tokens)}}
        \footnotesize
        \textit{Dump every manual, document, and tool output into one prompt.}
        \begin{itemize}\scriptsize
          \item \textbf{"Lost in the Middle"}: Models forget or ignore facts tucked in the middle paragraphs.
          \item \textbf{Prompt Pollution}: Intermediate tool errors and scrapings distract subsequent reasoning.
          \item \textbf{Strictly Sequential}: You cannot run pieces in parallel; long waiting times.
          \item \textbf{Impossible to Debug}: When it fails, you don't know which document confused it.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Idea B: Orchestrator-Workers}}
        \footnotesize
        \textit{Lead agent breaks the job down $\to$ dispatches isolated workers $\to$ combines answers.}
        \begin{itemize}\scriptsize
          \item \textbf{Context Isolation}: Each worker gets only what it needs $\to$ sharp, focused attention.
          \item \textbf{Parallel Execution}: Multiple subtasks run at the exact same second.
          \item \textbf{Custom Prompts}: A retrieval worker uses different instructions than a calculation worker.
          \item \textbf{Unit Testable}: You can cache and test each worker independently.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{Hung's Core Point on the "Orchestration Tax"}},fonttitle=\footnotesize\bfseries]
    \footnotesize Orchestrators cost extra planning tokens and lose cross-worker context. But they buy you \textbf{clean attention and parallelism} that a single mega-context cannot match.
  \end{tcolorbox}
\end{frame}

% SLIDE 11: THE LEAD/SUBAGENT PATTERN: BIG GAINS, BUT AT 8X THE BILL
\begin{frame}{The Lead/Subagent Pattern: Big Gains, but at 8x the Bill}
  \framesubtitle{Presenter: Vu Viet Hung $\vert$ Anthropic's Multi-Agent Research System (2025)}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Evaluator-Optimizer: The Review Loop}}
        \footnotesize
        \begin{itemize}
          \item Worker drafts $\to$ Evaluator checks $\to$ Worker rewrites.
          \item Stops when it meets clear criteria or hits a retry limit.
          \item \textbf{The Hidden Trap}: If the evaluator uses the same model weights as the worker, both share the exact same blind spots.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Lead/Subagents (Anthropic 2025)}}
        \footnotesize
        \textit{How Anthropic structured multi-agent research}:
        \begin{itemize}
          \item \textbf{Lead Agent}: Keeps the master plan, assigns subtasks, writes the final report.
          \item \textbf{Subagents}: Deep dive into narrow searches and return short summaries.
          \item \textbf{The Real Trade-Off}:
            \begin{itemize}\scriptsize
              \item \textbf{Capability}: Solves complex multi-hour tasks single agents fail at.
              \item \textbf{The Price}: $3\times - 8\times$ total token usage; high latency bottleneck at the Lead's synthesis step.
            \end{itemize}
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{block}{\textbf{Hung's Defense Talking Point}}
    \footnotesize Why did Anthropic reject peer-to-peer chats? Because letting subagents chat without a supervisor creates conversational sycophancy where agents agree with each other's mistakes.
  \end{block}
\end{frame}

% SLIDE 12: WHAT HAPPENS WHEN YOUR AGENT CRASHES 45 MINUTES IN?
\begin{frame}{What Happens When Your Agent Crashes 45 Minutes In?}
  \framesubtitle{Presenter: Vu Viet Hung $\vert$ LangGraph, Durable Execution \& Checkpointing}

  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \textbf{\color{hcmutblue}The Problem: Long Tasks Always Break}
      \begin{itemize}\footnotesize
        \item When an agent runs a 45-minute task, APIs drop, network connections reset, and rate limits hit.
        \item \textbf{Without Durable Execution}: The process crashes $\to$ state disappears $\to$ you start over from minute zero and re-pay for every token.
      \end{itemize}
      \vspace{0.1cm}
      \textbf{\color{hcmutblue}How LangGraph Solves This}:
      \begin{enumerate}\footnotesize
        \item \textbf{Explicit State Graphs}: State is a clean dictionary modified only through defined reducer functions.
        \item \textbf{Checkpointing}: After every step, a snapshot is saved directly to PostgreSQL or Redis.
        \item \textbf{Human-in-the-Loop}: Pause execution, wait for a human click, and resume seamlessly.
      \end{enumerate}
    \end{column}

    \begin{column}{0.47\textwidth}
      \begin{tcolorbox}[colback=boxbg,colframe=darkgreen,title={\textbf{Time-Travel Debugging}},fonttitle=\footnotesize\bfseries]
        \footnotesize
        \begin{itemize}
          \item \textbf{Rewind}: Roll state back to Step 4 after fixing a bad prompt.
          \item \textbf{Human Edits}: A human fixes a corrupted variable, and the agent continues on the new branch.
          \item \textbf{Instant Recovery}: When a server reboots, the agent picks up at the exact node it left off.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{alertblock}{\textbf{Hung's Takeaway}}
    \footnotesize Checkpointing isn't an optional devops trick; it is what turns an unreliable demo into production software.
  \end{alertblock}
\end{frame}

% SLIDE 13: WHY ASKING AN LLM TO "DOUBLE-CHECK" MAKES IT WORSE
\begin{frame}{Why Asking an LLM to "Double-Check" Usually Makes Things Worse}
  \framesubtitle{Presenter: Vu Viet Hung $\vert$ Huang et al. (2023): The Myth of Intrinsic Self-Correction}

  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \textbf{\color{darkred}What Huang et al. (ICLR 2024) Proved}:
      \begin{itemize}\footnotesize
        \item Across reasoning tests (math, logic, commonsense), \textbf{LLMs cannot fix reasoning errors without external tools}.
        \item When told: \textit{"Review your work and fix any mistakes"}:
          \begin{itemize}\scriptsize
            \item Fixing a Wrong Answer to Right: Barely happened ($< 5\%$).
            \item Changing a Right Answer to Wrong: Happened often ($15\% - 25\%$).
          \end{itemize}
        \item \textbf{Conclusion}: Blind self-correction \textbf{hurts overall accuracy}.
      \end{itemize}
    \end{column}

    \begin{column}{0.45\textwidth}
      \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Why Does This Happen?}},fonttitle=\footnotesize\bfseries]
        \footnotesize
        \begin{enumerate}
          \item \textbf{Same Brain, Same Blind Spot}: If the model didn't know the logic in turn 1, it doesn't suddenly discover it in turn 2.
          \item \textbf{Eager to Please}: When you ask "Are you sure?", the model assumes you spotted a mistake, panics, and second-guesses its correct answer.
        \end{enumerate}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{alertblock}{\textbf{Hung's Provocation for the Committee}}
    \footnotesize Why did older papers claim self-refinement worked? Because they tested prose writing (editing style) or accidentally leaked correct answers in the critique prompt!
  \end{alertblock}
\end{frame}

% SLIDE 16: WHERE DOES REAL FEEDBACK COME FROM?
\begin{frame}{The Crucial Question: Where Does the Feedback Actually Come From?}
  \framesubtitle{Presenter: Dang Lam Tung (Lead) $\vert$ Internal Imagination vs. External Ground Truth}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{1. Self-Generated Feedback (Internal)}}
        \footnotesize
        \textbf{What it is}: The model re-reads its own output and grades itself.\\
        \textbf{The problem}: It uses the exact same probability distribution that made the error.\\
        \textbf{What happens}:
        \begin{itemize}\scriptsize
          \item Confirms its own misconceptions.
          \item Generates believable excuses for wrong steps.
          \item Like a student grading their own test without an answer key.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{2. Environment Feedback (External Oracles)}}
        \footnotesize
        \textbf{What it is}: Signal returned by an external system that cannot be fooled.\\
        \textbf{Real-world examples}:
        \begin{itemize}\scriptsize
          \item Code execution (Python errors, compiler output).
          \item Database responses (Record exists / Revoked).
          \item Math solvers and formal verification engines.
          \item A human expert in the loop.
        \end{itemize}
        \textbf{Why it matters}: It is the only true anchor to reality.
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Tung's Thesis for Defense}},fonttitle=\footnotesize\bfseries]
    \footnotesize An agent cannot pull itself out of a hallucination by thinking harder. Without an \textbf{external verification tool}, iterative loops are pure theater.
  \end{tcolorbox}
\end{frame}

% SLIDE 17: CATCHING MISTAKES EARLY (PRM VS ORM)
\begin{frame}{Catching Mistakes Early: Why Step-by-Step Verification Beats Final Scores}
  \framesubtitle{Presenter: Dang Lam Tung (Lead) $\vert$ Lightman et al. (2023 - OpenAI): Let's Verify Step by Step}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Outcome Models: Scoring the Final Answer (ORM)}}
        \footnotesize
        \begin{itemize}
          \item Only checks if the final answer matches ($0$ or $1$).
          \item \textbf{The Fatal Flaw}: Rewards broken reasoning that accidentally got lucky.
          \item Cannot tell you which step went off the rails.
          \item Cannot guide search trees because credit is only assigned at the end.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Process Models: Scoring Every Single Step (PRM)}}
        \footnotesize
        \begin{itemize}
          \item Evaluates the validity of \textbf{each reasoning step} as it happens.
          \item \textbf{Why it changes everything}:
            \begin{itemize}\scriptsize
              \item Catches the exact moment logic fails.
              \item Enables tree search (MCTS) to prune bad branches immediately.
              \item Prevents the model from "faking" alignment.
            \end{itemize}
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=boxbg,colframe=hcmutblue,title={\textbf{What This Means for Agents}},fonttitle=\footnotesize\bfseries]
    \footnotesize Process supervision is how frontier models (like OpenAI o1/o3) scale inference-time compute. They don't just generate text; they search a path checked at every step.
  \end{tcolorbox}
\end{frame}

% SLIDE 18: TEACHING A MODEL BY KEEPING WHAT WORKS (STAR)
\begin{frame}{Teaching a Model to Reason by Keeping Only What It Gets Right}
  \framesubtitle{Presenter: Dang Lam Tung (Lead) $\vert$ STaR (Zelikman et al., 2022): Bootstrapping Reasoning}

  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{The STaR Bootstrap Loop}}
        \footnotesize
        \begin{enumerate}
          \item \textbf{Generate}: Model tries to solve a problem with step-by-step thinking.
          \item \textbf{Filter (The Crucial Gate)}: Compare the final answer with the \textbf{Ground-Truth Answer Key}. \textit{Throw out everything that got it wrong.}
          \item \textbf{Rationalize}: If it failed, give it the right answer and let it re-trace the logic backward.
          \item \textbf{Finetune}: Train the model only on the reasoning paths that proved correct.
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.47\textwidth}
      \begin{tcolorbox}[colback=boxbg,colframe=darkred,title={\textbf{Notice Where the Signal Comes From}},fonttitle=\footnotesize\bfseries]
        \footnotesize
        \begin{itemize}
          \item STaR does \textbf{not} magically teach itself.
          \item The entire improvement relies on the \textbf{Answer Key in Step 2}.
          \item If you remove ground truth and let the model pick what it "likes", its reasoning collapses into self-reinforcing errors.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{block}{\textbf{Tung's Insight}}
    \footnotesize Self-training is impossible in a vacuum. You always need an external ground-truth filter to separate signal from noise.
  \end{block}
\end{frame}
